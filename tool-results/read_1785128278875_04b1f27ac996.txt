     2→-- Migration 004: Checkout, Orders, Payments, Rewards, Branches
     3→-- ----------------------------------------------------------------------------
     4→-- Phase 2 - Module 3: Production-grade checkout & payment system.
     5→--
     6→-- Design principles (carried over from migrations 001-003):
     7→--  1. All money stored as INTEGER PAISE (1 INR = 100 paise) — never floats.
     8→--  2. All weights stored as INTEGER GRAMS.
     9→--  3. RLS enabled on every table. Users can ONLY access their own rows.
    10→--  4. All mutations go through SECURITY DEFINER RPC functions which
    11→--     perform server-side validation. Users NEVER directly mutate these
    12→--     tables (except reads on branches, which is public).
    13→--  5. Idempotency: every order has a unique idempotency_key; every webhook
    14→--     event is deduplicated via processed_webhook_events table.
    15→--  6. Append-only reward_transactions ledger + denormalized reward_balance
    16→--     for O(1) reads. Balance is updated atomically inside the same
    17→--     transaction that inserts the ledger row — no drift possible.
    18→--  7. Future-ready: orders table reserves pickup_pin and admin_assigned_to
    19→--     columns for later modules (no schema changes needed).
    20→--
    21→-- Timezone handling:
    22→--  - Supabase Postgres stores timestamptz in UTC internally.
    23→--  - Pickup date is DATE (no tz). Slot start/end are TIME (no tz).
    24→--  - "Today" is computed at the APPLICATION layer using Asia/Kolkata
    25→--    (IST) and validated against the requested pickup_date.

   522→-- Rules: 10 points = ₹5 (500 paise). Must be multiple of 10.
   523→-- ============================================================================
   524→create or replace function public.preview_reward_redemption(
   525→  p_user_id uuid,
   526→  p_points integer
   527→)
   528→returns jsonb
   529→language plpgsql

   533→declare
   534→  v_balance integer;
   535→  v_discount_paise integer;

   538→    return jsonb_build_object('success', false, 'message', 'Invalid points value');
   539→  end if;
   540→
   541→  if p_points % 10 <> 0 then
   542→    return jsonb_build_object('success', false, 'message', 'Points must be a multiple of 10');
   543→  end if;
   544→
   545→  if p_points < 10 then
   546→    return jsonb_build_object('success', false, 'message', 'Minimum 10 points required');
   547→  end if;
   548→
   549→  -- Get current balance (lazy init)
   550→  insert into public.reward_balance (user_id)
   551→  values (p_user_id)
   552→  on conflict (user_id) do nothing;
   553→
   554→  select balance_points into v_balance
   555→  from public.reward_balance
   556→  where user_id = p_user_id
   557→  for update;  -- lock row for the duration of this transaction
   558→
   559→  if v_balance is null then
   560→    return jsonb_build_object('success', false, 'message', 'Balance not found');
   561→  end if;
   562→
   563→  if p_points > v_balance then
   564→    return jsonb_build_object(
   565→      'success', false,
   566→      'message', 'Insufficient points',
   567→      'available', v_balance
   568→    );
   569→  end if;
   570→
   571→  -- 10 points = 500 paise → discount = (points / 10) * 500
   572→  v_discount_paise := (p_points / 10) * 500;
   573→
   574→  return jsonb_build_object(
   575→    'success', true,
   576→    'points', p_points,
   577→    'discount_paise', v_discount_paise,
   578→    'balance_after', v_balance - p_points
   579→  );
   580→end;
   581→$$;

   585→-- ============================================================================
   586→-- RPC: create_draft_order
   587→-- ----------------------------------------------------------------------------
   588→-- Atomic creation of:
   589→--   1. Order row (status='draft', payment_status='pending')
   590→--   2. Order items rows (from p_items JSON array)
   591→--   3. Reward point deduction (if p_reward_points_to_redeem > 0)
   592→--      - Locks reward_balance row
   593→--      - Verifies sufficient balance
   594→--      - Deducts + inserts reward_transactions ledger row
   595→--
   596→-- Idempotency:
   597→--   - p_idempotency_key is UNIQUE. If the same key is submitted twice,
   598→--     the second call fails with a unique constraint violation, which
   599→--     the application layer catches and returns the original order.
   600→--
   601→-- Item JSON shape:
   602→--   [{line_key, item_id, variant_id, item_name, item_emoji, item_type,
   603→--     variant_label, weight_grams, piece_count, unit_price_paise,
   604→--     quantity, line_total_paise}, ...]
   605→--
   606→-- ALL prices are recomputed at the APPLICATION layer (cart-service) before
   607→-- this function is called. The function still enforces:
   608→--   - subtotal = SUM(line_total_paise) for the order's items
   609→--   - final_amount = subtotal + donations - reward_discount
   610→--   - reward_points_to_redeem is multiple of 10 and <= balance
   611→-- ============================================================================
   612→create or replace function public.create_draft_order(
   613→  p_user_id uuid,
   614→  p_branch_id uuid,
   615→  p_pickup_date date,
   616→  p_slot_start time,
   617→  p_slot_end time,
   618→  p_subtotal_paise integer,
   619→  p_donation_plantation_paise integer,
   620→  p_donation_hunger_paise integer,
   621→  p_reward_points_to_redeem integer,
   622→  p_reward_discount_paise integer,
   623→  p_final_amount_paise integer,
   624→  p_idempotency_key text,
   625→  p_items jsonb
   626→)
   627→returns jsonb
   628→language plpgsql
   629→security definer
   630→set search_path = public
   631→as $$
   632→declare
   633→  v_order_id uuid;
   634→  v_order_number text;
   635→  v_balance integer;
   636→  v_new_balance integer;
   637→  v_computed_subtotal integer := 0;
   638→  v_computed_final integer;
   639→  v_item jsonb;
   640→  v_expected_discount integer;
   641→  v_expected_final integer;
   642→begin
   643→  -- ---------------- VALIDATION ----------------
   644→  if p_subtotal_paise < 0 then
   645→    return jsonb_build_object('success', false, 'message', 'Invalid subtotal');
   646→  end if;
   647→
   648→  if p_donation_plantation_paise not in (0, 500) then
   649→    return jsonb_build_object('success', false, 'message', 'Invalid plantation donation');
   650→  end if;
   651→
   652→  if p_donation_hunger_paise not in (0, 1000) then
   653→    return jsonb_build_object('success', false, 'message', 'Invalid hunger donation');
   654→  end if;
   655→
   656→  if p_reward_points_to_redeem < 0 or p_reward_points_to_redeem % 10 <> 0 then
   657→    return jsonb_build_object('success', false, 'message', 'Invalid reward points');
   658→  end if;
   659→
   660→  if p_reward_discount_paise < 0 then
   661→    return jsonb_build_object('success', false, 'message', 'Invalid reward discount');
   662→  end if;
   663→
   664→  if p_final_amount_paise < 0 then
   665→    return jsonb_build_object('success', false, 'message', 'Invalid final amount');
   666→  end if;
   667→
   668→  -- Verify reward discount matches points redeemed
   669→  if p_reward_points_to_redeem > 0 then
   670→    v_expected_discount := (p_reward_points_to_redeem / 10) * 500;
   671→    if p_reward_discount_paise <> v_expected_discount then
   672→      return jsonb_build_object('success', false, 'message', 'Reward discount mismatch');
   673→    end if;
   674→  else
   675→    if p_reward_discount_paise <> 0 then
   676→      return jsonb_build_object('success', false, 'message', 'Reward discount without redemption');
   677→    end if;
   678→  end if;
   679→
   680→  -- Verify final amount = subtotal + donations - reward_discount
   681→  v_expected_final := p_subtotal_paise + p_donation_plantation_paise
   682→                      + p_donation_hunger_paise - p_reward_discount_paise;
   683→  if p_final_amount_paise <> v_expected_final then
   684→    return jsonb_build_object('success', false, 'message', 'Final amount mismatch');
   685→  end if;
   686→
   687→  -- If redeeming points: verify balance + deduct atomically
   688→  if p_reward_points_to_redeem > 0 then
   689→    -- Lazy init balance row
   690→    insert into public.reward_balance (user_id)
   691→    values (p_user_id)
   692→    on conflict (user_id) do nothing;
   693→
   694→    select balance_points into v_balance
   695→    from public.reward_balance
   696→    where user_id = p_user_id
   697→    for update;  -- row lock
   698→
   699→    if v_balance is null then
   700→      return jsonb_build_object('success', false, 'message', 'Reward balance not found');
   701→    end if;
   702→
   703→    if p_reward_points_to_redeem > v_balance then
   704→      return jsonb_build_object(
   705→        'success', false,
   706→        'message', 'Insufficient reward points',
   707→        'available', v_balance
   708→      );
   709→    end if;
   710→
   711→    v_new_balance := v_balance - p_reward_points_to_redeem;
   712→
   713→    update public.reward_balance
   714→    set balance_points = v_new_balance,
   715→        total_redeemed = total_redeemed + p_reward_points_to_redeem
   716→    where user_id = p_user_id;
   717→  end if;
   718→
   719→  -- ---------------- CREATE ORDER ----------------
   720→  v_order_number := public.generate_order_number();
   721→
   722→  insert into public.orders (
   723→    order_number, user_id, branch_id,
   724→    pickup_date, pickup_slot_start, pickup_slot_end,
   725→    subtotal_paise,
   726→    donation_plantation_paise, donation_hunger_paise,
   727→    reward_points_redeemed, reward_discount_paise,
   728→    final_amount_paise,
   729→    payment_status, order_status,
   730→    idempotency_key
   731→  ) values (
   732→    v_order_number, p_user_id, p_branch_id,
   733→    p_pickup_date, p_slot_start, p_slot_end,
   734→    p_subtotal_paise,
   735→    p_donation_plantation_paise, p_donation_hunger_paise,
   736→    p_reward_points_to_redeem, p_reward_discount_paise,
   737→    p_final_amount_paise,
   738→    'pending', 'draft',
   739→    p_idempotency_key
   740→  )
   741→  returning id into v_order_id;
   742→
   743→  -- ---------------- INSERT ITEMS ----------------
   744→  if jsonb_array_length(p_items) = 0 then
   745→    raise exception 'Cannot create order with empty items';
   746→  end if;
   747→
   748→  for v_item in select * from jsonb_array_elements(p_items) loop
   749→    -- Accumulate subtotal for verification
   750→    v_computed_subtotal := v_computed_subtotal + (v_item->>'line_total_paise')::integer;
   751→
   752→    insert into public.order_items (
   753→      order_id, user_id,
   754→      line_key, item_id, variant_id,
   755→      item_name, item_emoji, item_type, variant_label,
   756→      weight_grams, piece_count,
   757→      unit_price_paise, quantity, line_total_paise
   758→    ) values (
   759→      v_order_id, p_user_id,
   760→      v_item->>'line_key',
   761→      v_item->>'item_id',
   762→      v_item->>'variant_id',
   763→      v_item->>'item_name',
   764→      v_item->>'item_emoji',
   765→      v_item->>'item_type',
   766→      v_item->>'variant_label',
   767→      nullif(v_item->>'weight_grams', '')::integer,
   768→      nullif(v_item->>'piece_count', '')::integer,
   769→      (v_item->>'unit_price_paise')::integer,
   770→      (v_item->>'quantity')::integer,
   771→      (v_item->>'line_total_paise')::integer
   772→    );
   773→  end loop;
   774→
   775→  -- Verify computed subtotal matches declared subtotal
   776→  if v_computed_subtotal <> p_subtotal_paise then
   777→    raise exception 'Subtotal mismatch: computed %, declared %',
   778→      v_computed_subtotal, p_subtotal_paise;
   779→  end if;
   780→
   781→  -- ---------------- REWARD LEDGER (REDEEM) ----------------
   782→  if p_reward_points_to_redeem > 0 then
   783→    insert into public.reward_transactions (
   784→      user_id, order_id, points, type, reason, balance_after
   785→    ) values (
   786→      p_user_id, v_order_id,
   787→      -p_reward_points_to_redeem,
   788→      'redeem',
   789→      'Redeemed for order ' || v_order_number,
   790→      v_new_balance
   791→    );
   792→  end if;
   793→
   794→  return jsonb_build_object(
   795→    'success', true,
   796→    'order_id', v_order_id,
   797→    'order_number', v_order_number
   798→  );
   799→end;
   800→$$;

   806→-- ============================================================================
   807→-- RPC: attach_payment_intent_to_order
   808→-- ----------------------------------------------------------------------------
   809→-- Links a Stripe PaymentIntent ID to a draft order and transitions the
   810→-- order to 'awaiting_payment' status. Also creates the payments row.
   811→-- Idempotent: if called twice with same PI ID, returns success without
   812→-- creating duplicates.
   813→-- ============================================================================
   814→create or replace function public.attach_payment_intent_to_order(
   815→  p_order_id uuid,
   816→  p_user_id uuid,
   817→  p_stripe_payment_intent_id text,
   818→  p_amount_paise integer
   819→)
   820→returns jsonb
   821→language plpgsql
   822→security definer
   823→set search_path = public
   824→as $$
   825→declare
   826→  v_existing_pi text;
   827→  v_payment_id uuid;
   828→begin
   829→  -- Lock the order row
   830→  select stripe_payment_intent_id into v_existing_pi
   831→  from public.orders
   832→  where id = p_order_id and user_id = p_user_id
   833→  for update;
   834→
   835→  if not found then
   836→    return jsonb_build_object('success', false, 'message', 'Order not found');
   837→  end if;
   838→
   839→  -- If already attached to a different PI, reject
   840→  if v_existing_pi is not null and v_existing_pi <> p_stripe_payment_intent_id then
   841→    return jsonb_build_object('success', false, 'message', 'Order already has a different payment intent');
   842→  end if;
   843→
   844→  -- Update order to awaiting_payment (if not already)
   845→  update public.orders
   846→  set order_status = case when order_status = 'draft' then 'awaiting_payment' else order_status end,
   847→      stripe_payment_intent_id = p_stripe_payment_intent_id
   848→  where id = p_order_id;
   849→
   850→  -- Insert payment row (idempotent: if exists for this PI, do nothing)
   851→  insert into public.payments (
   852→    order_id, user_id,
   853→    stripe_payment_intent_id, amount_paise, currency, status
   854→  ) values (
   855→    p_order_id, p_user_id,
   856→    p_stripe_payment_intent_id, p_amount_paise, 'inr', 'pending'
   857→  )
   858→  on conflict (stripe_payment_intent_id) do nothing
   859→  returning id into v_payment_id;
   860→
   861→  -- If conflict (existing payment), fetch its id
   862→  if v_payment_id is null then
   863→    select id into v_payment_id
   864→    from public.payments
   865→    where stripe_payment_intent_id = p_stripe_payment_intent_id
   866→    limit 1;
   867→  end if;
   868→
   869→  return jsonb_build_object(
   870→    'success', true,
   871→    'order_id', p_order_id,
   872→    'payment_id', v_payment_id
   873→  );
   874→end;
   875→$$;

   887→-- Atomic operations:

   894→-- If called twice with same event_id, second call is a no-op (success).

   897→  p_order_id uuid,
   898→  p_stripe_charge_id text,
   899→  p_webhook_event_id text,
   900→  p_event_type text,
   901→  p_raw_payload jsonb

   909→  v_order public.orders%rowtype;
   910→  v_payment_id uuid;
   911→  v_balance integer;
   912→  v_new_balance integer;
   913→  v_points_to_earn integer := 0;
   914→  v_user_id uuid;
   915→begin
   916→  -- ---------------- IDEMPOTENCY CHECK ----------------
   917→  -- Try to record this event. If it already exists, we've processed it.
   918→  begin
   919→    insert into public.processed_webhook_events (event_id, event_type, payment_id)
   920→    values (p_webhook_event_id, p_event_type, null)
   921→    returning payment_id into v_payment_id;

   923→    return jsonb_build_object('success', true, 'message', 'Event already processed', 'idempotent', true);
   924→  end;
   925→
   926→  -- ---------------- LOAD ORDER (lock) ----------------
   927→  select * into v_order
   928→  from public.orders
   929→  where id = p_order_id
   930→  for update;
   931→
   932→  if not found then
   933→    return jsonb_build_object('success', false, 'message', 'Order not found');
   934→  end if;
   935→
   936→  v_user_id := v_order.user_id;
   937→

   939→  select id into v_payment_id
   940→  from public.payments
   941→  where stripe_payment_intent_id = v_order.stripe_payment_intent_id
   942→  limit 1;
   943→

   945→  if v_payment_id is not null then
   946→    update public.payments
   947→    set status = 'succeeded',
   948→        stripe_charge_id = p_stripe_charge_id,
   949→        raw_payload = p_raw_payload,
   950→        webhook_processed_at = now()
   951→    where id = v_payment_id;
   952→
   953→    -- Backfill payment_id in processed_webhook_events
   954→    update public.processed_webhook_events
   955→    set payment_id = v_payment_id
   956→    where event_id = p_webhook_event_id;
   957→  end if;

   960→  if v_order.order_status = 'confirmed' then
   961→    return jsonb_build_object('success', true, 'message', 'Order already confirmed', 'idempotent', true);
   962→  end if;
   963→
   964→  -- ---------------- UPDATE ORDER STATUS ----------------
   965→  update public.orders
   966→  set payment_status = 'succeeded',
   967→      order_status = 'confirmed'
   968→  where id = p_order_id;
   969→
   970→  -- ---------------- AWARD REWARD POINTS ----------------
   971→  -- Rule: 5 points if BOTH conditions true:
   972→  --   1. subtotal_paise > 50000 (₹500)
   973→  --   2. donation_plantation_paise = 500 (₹5)
   974→  if v_order.subtotal_paise > 50000 and v_order.donation_plantation_paise = 500 then
   975→    v_points_to_earn := 5;
   976→  end if;
   977→
   978→  if v_points_to_earn > 0 then
   979→    -- Lazy init balance row
   980→    insert into public.reward_balance (user_id)
   981→    values (v_user_id)
   982→    on conflict (user_id) do nothing;
   983→
   984→    select balance_points into v_balance
   985→    from public.reward_balance
   986→    where user_id = v_user_id
   987→    for update;
   988→
   989→    v_new_balance := v_balance + v_points_to_earn;
   990→
   991→    update public.reward_balance
   992→    set balance_points = v_new_balance,
   993→        total_earned = total_earned + v_points_to_earn
   994→    where user_id = v_user_id;
   995→
   996→    insert into public.reward_transactions (
   997→      user_id, order_id, points, type, reason, balance_after
   998→    ) values (
   999→      v_user_id, p_order_id,
  1000→      v_points_to_earn,
  1001→      'earn',
  1002→      'Earned from order ' || v_order.order_number,
  1003→      v_new_balance
  1004→    );
  1005→
  1006→    -- Record earned points on the order
  1007→    update public.orders
  1008→    set reward_points_earned = v_points_to_earn
  1009→    where id = p_order_id;
  1010→  end if;
  1011→
  1012→  -- ---------------- CLEAR USER'S CART ----------------
  1013→  -- Cart is now an order — empty the plate.
  1014→  delete from public.cart_items where user_id = v_user_id;
  1015→
  1016→  return jsonb_build_object(
  1017→    'success', true,
  1018→    'order_id', p_order_id,
  1019→    'order_number', v_order.order_number,
  1020→    'reward_points_earned', v_points_to_earn
  1021→  );
  1022→end;

  1096→  -- If order is already failed/confirmed, skip
  1097→  if v_order.order_status in ('failed', 'confirmed') then
  1098→    return jsonb_build_object('success', true, 'message', 'Order already in terminal state', 'idempotent', true);
  1099→  end if;
  1100→
  1101→  -- ---------------- UPDATE ORDER STATUS ----------------
  1102→  update public.orders
  1103→  set payment_status = 'failed',
  1104→      order_status = 'failed'
  1105→  where id = p_order_id;
  1106→
  1107→  -- ---------------- RESTORE REDEEMED POINTS ----------------
  1108→  if v_order.reward_points_redeemed > 0 then
  1109→    select balance_points into v_balance
  1110→    from public.reward_balance
  1111→    where user_id = v_user_id
  1112→    for update;
  1113→
  1114→    v_new_balance := v_balance + v_order.reward_points_redeemed;
  1115→
  1116→    update public.reward_balance
  1117→    set balance_points = v_new_balance,
  1118→        total_redeemed = greatest(total_redeemed - v_order.reward_points_redeemed, 0)
  1119→    where user_id = v_user_id;
  1120→
  1121→    insert into public.reward_transactions (
  1122→      user_id, order_id, points, type, reason, balance_after
  1123→    ) values (
  1124→      v_user_id, p_order_id,
  1125→      v_order.reward_points_redeemed,
  1126→      'restore',
  1127→      'Restored from failed order ' || v_order.order_number,
  1128→      v_new_balance
  1129→    );
  1130→  end if;
  1131→
  1132→  return jsonb_build_object(
  1133→    'success', true,
  1134→    'order_id', p_order_id,
  1135→    'order_number', v_order.order_number,
  1136→    'reward_points_restored', v_order.reward_points_redeemed
  1137→  );
  1138→end;

  1146→-- Returns the full order (header + items) for a user. Only the order's

  1222→-- RPC: cancel_draft_order
  1223→-- ----------------------------------------------------------------------------
  1224→-- Allows the user to cancel an order that's still in 'draft' or
  1225→-- 'awaiting_payment' state (e.g. they abandoned checkout). Restores
  1226→-- redeemed points. Once 'confirmed' or 'failed', cannot be cancelled.
  1227→-- ============================================================================
  1228→create or replace function public.cancel_draft_order(
  1229→  p_user_id uuid,
  1230→  p_order_id uuid
  1231→)
  1232→returns jsonb
  1233→language plpgsql
  1234→security definer
  1235→set search_path = public

  1238→  v_order public.orders%rowtype;
  1239→  v_balance integer;
  1240→  v_new_balance integer;

  1242→  select * into v_order
  1243→  from public.orders
  1244→  where id = p_order_id and user_id = p_user_id
  1245→  for update;
  1246→

  1248→    return jsonb_build_object('success', false, 'message', 'Order not found');

  1252→    return jsonb_build_object(

  1254→      'message', 'Cannot cancel order in current state: ' || v_order.order_status

  1256→  end if;
  1257→
  1258→  update public.orders
  1259→  set order_status = 'cancelled'
  1260→  where id = p_order_id;
  1261→
  1262→  -- Restore redeemed points
  1263→  if v_order.reward_points_redeemed > 0 then
  1264→    select balance_points into v_balance
  1265→    from public.reward_balance
  1266→    where user_id = p_user_id

  1268→
  1269→    v_new_balance := v_balance + v_order.reward_points_redeemed;
  1270→
  1271→    update public.reward_balance
  1272→    set balance_points = v_new_balance
  1273→    where user_id = p_user_id;

  1279→      v_order.reward_points_redeemed,
  1280→      'restore',
  1281→      'Restored from cancelled order ' || v_order.order_number,
  1282→      v_new_balance

  1286→  return jsonb_build_object('success', true, 'order_id', p_order_id);
