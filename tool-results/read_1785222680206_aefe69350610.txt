     3→     3  -- ----------------------------------------------------------------------------
     4→     4  -- Phase 2 - Module 3: Production-grade checkout & payment system.
     5→     5  --
     6→     6  -- Design principles (carried over from migrations 001-003):
     7→     7  --  1. All money stored as INTEGER PAISE (1 INR = 100 paise) — never floats.
     8→     8  --  2. All weights stored as INTEGER GRAMS.
     9→     9  --  3. RLS enabled on every table. Users can ONLY access their own rows.
    10→    10  --  4. All mutations go through SECURITY DEFINER RPC functions which
    11→    11  --     perform server-side validation. Users NEVER directly mutate these
    12→    12  --     tables (except reads on branches, which is public).
    13→    13  --  5. Idempotency: every order has a unique idempotency_key; every webhook
    14→    14  --     event is deduplicated via processed_webhook_events table.
    15→    15  --  6. Append-only reward_transactions ledger + denormalized reward_balance
    16→    16  --     for O(1) reads. Balance is updated atomically inside the same
    17→    17  --     transaction that inserts the ledger row — no drift possible.
    18→    18  --  7. Future-ready: orders table reserves pickup_pin and admin_assigned_to
    19→    19  --     columns for later modules (no schema changes needed).
    20→    20  --
    21→    21  -- Timezone handling:
    22→    22  --  - Supabase Postgres stores timestamptz in UTC internally.
    23→    23  --  - Pickup date is DATE (no tz). Slot start/end are TIME (no tz).
    24→    24  --  - "Today" is computed at the APPLICATION layer using Asia/Kolkata
    25→    25  --    (IST) and validated against the requested pickup_date.

   518→   518  -- ============================================================================
   519→   519  -- RPC: preview_reward_redemption
   520→   520  -- ----------------------------------------------------------------------------
   521→   521  -- Returns the discount (paise) the user would get for redeeming N points,
   522→   522  -- WITHOUT actually deducting. Pure calc with validation.
   523→   523  -- Rules: 10 points = ₹5 (500 paise). Must be multiple of 10.
   524→   524  -- ============================================================================
   525→   525  create or replace function public.preview_reward_redemption(
   526→   526    p_user_id uuid,
   527→   527    p_points integer
   528→   528  )
   529→   529  returns jsonb
   530→   530  language plpgsql
   531→   531  security definer
   532→   532  set search_path = public
   533→   533  as $$
   534→   534  declare
   535→   535    v_balance integer;
   536→   536    v_discount_paise integer;
   537→   537  begin
   538→   538    if p_points is null or p_points < 0 then
   539→   539      return jsonb_build_object('success', false, 'message', 'Invalid points value');
   540→   540    end if;
   541→   541  
   542→   542    if p_points % 10 <> 0 then
   543→   543      return jsonb_build_object('success', false, 'message', 'Points must be a multiple of 10');
   544→   544    end if;
   545→   545  
   546→   546    if p_points < 10 then
   547→   547      return jsonb_build_object('success', false, 'message', 'Minimum 10 points required');
   548→   548    end if;
   549→   549  
   550→   550    -- Get current balance (lazy init)
   551→   551    insert into public.reward_balance (user_id)
   552→   552    values (p_user_id)
   553→   553    on conflict (user_id) do nothing;
   554→   554  
   555→   555    select balance_points into v_balance
   556→   556    from public.reward_balance
   557→   557    where user_id = p_user_id
   558→   558    for update;  -- lock row for the duration of this transaction
   559→   559  
   560→   560    if v_balance is null then
   561→   561      return jsonb_build_object('success', false, 'message', 'Balance not found');
   562→   562    end if;
   563→   563  
   564→   564    if p_points > v_balance then
   565→   565      return jsonb_build_object(
   566→   566        'success', false,
   567→   567        'message', 'Insufficient points',
   568→   568        'available', v_balance
   569→   569      );
   570→   570    end if;
   571→   571  
   572→   572    -- 10 points = 500 paise → discount = (points / 10) * 500
   573→   573    v_discount_paise := (p_points / 10) * 500;
   574→   574  
   575→   575    return jsonb_build_object(
   576→   576      'success', true,
   577→   577      'points', p_points,
   578→   578      'discount_paise', v_discount_paise,
   579→   579      'balance_after', v_balance - p_points
   580→   580    );
   581→   581  end;
   582→   582  $$;
   583→   583  
   584→   584  grant execute on function public.preview_reward_redemption(uuid, integer) to authenticated;

   586→   586  -- ============================================================================
   587→   587  -- RPC: create_draft_order
   588→   588  -- ----------------------------------------------------------------------------
   589→   589  -- Atomic creation of:
   590→   590  --   1. Order row (status='draft', payment_status='pending')
   591→   591  --   2. Order items rows (from p_items JSON array)
   592→   592  --   3. Reward point deduction (if p_reward_points_to_redeem > 0)
   593→   593  --      - Locks reward_balance row
   594→   594  --      - Verifies sufficient balance
   595→   595  --      - Deducts + inserts reward_transactions ledger row
   596→   596  --
   597→   597  -- Idempotency:
   598→   598  --   - p_idempotency_key is UNIQUE. If the same key is submitted twice,
   599→   599  --     the second call fails with a unique constraint violation, which
   600→   600  --     the application layer catches and returns the original order.
   601→   601  --
   602→   602  -- Item JSON shape:
   603→   603  --   [{line_key, item_id, variant_id, item_name, item_emoji, item_type,
   604→   604  --     variant_label, weight_grams, piece_count, unit_price_paise,
   605→   605  --     quantity, line_total_paise}, ...]
   606→   606  --
   607→   607  -- ALL prices are recomputed at the APPLICATION layer (cart-service) before
   608→   608  -- this function is called. The function still enforces:
   609→   609  --   - subtotal = SUM(line_total_paise) for the order's items
   610→   610  --   - final_amount = subtotal + donations - reward_discount
   611→   611  --   - reward_points_to_redeem is multiple of 10 and <= balance
   612→   612  -- ============================================================================
   613→   613  create or replace function public.create_draft_order(
   614→   614    p_user_id uuid,
   615→   615    p_branch_id uuid,
   616→   616    p_pickup_date date,
   617→   617    p_slot_start time,
   618→   618    p_slot_end time,
   619→   619    p_subtotal_paise integer,
   620→   620    p_donation_plantation_paise integer,
   621→   621    p_donation_hunger_paise integer,
   622→   622    p_reward_points_to_redeem integer,
   623→   623    p_reward_discount_paise integer,
   624→   624    p_final_amount_paise integer,
   625→   625    p_idempotency_key text,
   626→   626    p_items jsonb
   627→   627  )
   628→   628  returns jsonb
   629→   629  language plpgsql
   630→   630  security definer
   631→   631  set search_path = public
   632→   632  as $$
   633→   633  declare
   634→   634    v_order_id uuid;
   635→   635    v_order_number text;
   636→   636    v_balance integer;
   637→   637    v_new_balance integer;
   638→   638    v_computed_subtotal integer := 0;
   639→   639    v_computed_final integer;
   640→   640    v_item jsonb;
   641→   641    v_expected_discount integer;
   642→   642    v_expected_final integer;
   643→   643  begin
   644→   644    -- ---------------- VALIDATION ----------------
   645→   645    if p_subtotal_paise < 0 then
   646→   646      return jsonb_build_object('success', false, 'message', 'Invalid subtotal');
   647→   647    end if;
   648→   648  
   649→   649    if p_donation_plantation_paise not in (0, 500) then
   650→   650      return jsonb_build_object('success', false, 'message', 'Invalid plantation donation');
   651→   651    end if;
   652→   652  
   653→   653    if p_donation_hunger_paise not in (0, 1000) then
   654→   654      return jsonb_build_object('success', false, 'message', 'Invalid hunger donation');
   655→   655    end if;
   656→   656  
   657→   657    if p_reward_points_to_redeem < 0 or p_reward_points_to_redeem % 10 <> 0 then
   658→   658      return jsonb_build_object('success', false, 'message', 'Invalid reward points');
   659→   659    end if;
   660→   660  
   661→   661    if p_reward_discount_paise < 0 then
   662→   662      return jsonb_build_object('success', false, 'message', 'Invalid reward discount');
   663→   663    end if;
   664→   664  
   665→   665    if p_final_amount_paise < 0 then
   666→   666      return jsonb_build_object('success', false, 'message', 'Invalid final amount');
   667→   667    end if;
   668→   668  
   669→   669    -- Verify reward discount matches points redeemed
   670→   670    if p_reward_points_to_redeem > 0 then
   671→   671      v_expected_discount := (p_reward_points_to_redeem / 10) * 500;
   672→   672      if p_reward_discount_paise <> v_expected_discount then
   673→   673        return jsonb_build_object('success', false, 'message', 'Reward discount mismatch');
   674→   674      end if;
   675→   675    else
   676→   676      if p_reward_discount_paise <> 0 then
   677→   677        return jsonb_build_object('success', false, 'message', 'Reward discount without redemption');
   678→   678      end if;
   679→   679    end if;
   680→   680  
   681→   681    -- Verify final amount = subtotal + donations - reward_discount
   682→   682    v_expected_final := p_subtotal_paise + p_donation_plantation_paise
   683→   683                        + p_donation_hunger_paise - p_reward_discount_paise;
   684→   684    if p_final_amount_paise <> v_expected_final then
   685→   685      return jsonb_build_object('success', false, 'message', 'Final amount mismatch');
   686→   686    end if;
   687→   687  
   688→   688    -- If redeeming points: verify balance + deduct atomically
   689→   689    if p_reward_points_to_redeem > 0 then
   690→   690      -- Lazy init balance row
   691→   691      insert into public.reward_balance (user_id)
   692→   692      values (p_user_id)
   693→   693      on conflict (user_id) do nothing;
   694→   694  
   695→   695      select balance_points into v_balance
   696→   696      from public.reward_balance
   697→   697      where user_id = p_user_id
   698→   698      for update;  -- row lock
   699→   699  
   700→   700      if v_balance is null then
   701→   701        return jsonb_build_object('success', false, 'message', 'Reward balance not found');
   702→   702      end if;
   703→   703  
   704→   704      if p_reward_points_to_redeem > v_balance then
   705→   705        return jsonb_build_object(
   706→   706          'success', false,
   707→   707          'message', 'Insufficient reward points',
   708→   708          'available', v_balance
   709→   709        );
   710→   710      end if;
   711→   711  
   712→   712      v_new_balance := v_balance - p_reward_points_to_redeem;
   713→   713  
   714→   714      update public.reward_balance
   715→   715      set balance_points = v_new_balance,
   716→   716          total_redeemed = total_redeemed + p_reward_points_to_redeem
   717→   717      where user_id = p_user_id;
   718→   718    end if;
   719→   719  
   720→   720    -- ---------------- CREATE ORDER ----------------
   721→   721    v_order_number := public.generate_order_number();
   722→   722  
   723→   723    insert into public.orders (
   724→   724      order_number, user_id, branch_id,
   725→   725      pickup_date, pickup_slot_start, pickup_slot_end,
   726→   726      subtotal_paise,
   727→   727      donation_plantation_paise, donation_hunger_paise,
   728→   728      reward_points_redeemed, reward_discount_paise,
   729→   729      final_amount_paise,
   730→   730      payment_status, order_status,
   731→   731      idempotency_key
   732→   732    ) values (
   733→   733      v_order_number, p_user_id, p_branch_id,
   734→   734      p_pickup_date, p_slot_start, p_slot_end,
   735→   735      p_subtotal_paise,
   736→   736      p_donation_plantation_paise, p_donation_hunger_paise,
   737→   737      p_reward_points_to_redeem, p_reward_discount_paise,
   738→   738      p_final_amount_paise,
   739→   739      'pending', 'draft',
   740→   740      p_idempotency_key
   741→   741    )
   742→   742    returning id into v_order_id;
   743→   743  
   744→   744    -- ---------------- INSERT ITEMS ----------------
   745→   745    if jsonb_array_length(p_items) = 0 then
   746→   746      raise exception 'Cannot create order with empty items';
   747→   747    end if;
   748→   748  
   749→   749    for v_item in select * from jsonb_array_elements(p_items) loop
   750→   750      -- Accumulate subtotal for verification
   751→   751      v_computed_subtotal := v_computed_subtotal + (v_item->>'line_total_paise')::integer;
   752→   752  
   753→   753      insert into public.order_items (
   754→   754        order_id, user_id,
   755→   755        line_key, item_id, variant_id,
   756→   756        item_name, item_emoji, item_type, variant_label,
   757→   757        weight_grams, piece_count,
   758→   758        unit_price_paise, quantity, line_total_paise
   759→   759      ) values (
   760→   760        v_order_id, p_user_id,
   761→   761        v_item->>'line_key',
   762→   762        v_item->>'item_id',
   763→   763        v_item->>'variant_id',
   764→   764        v_item->>'item_name',
   765→   765        v_item->>'item_emoji',
   766→   766        v_item->>'item_type',
   767→   767        v_item->>'variant_label',
   768→   768        nullif(v_item->>'weight_grams', '')::integer,
   769→   769        nullif(v_item->>'piece_count', '')::integer,
   770→   770        (v_item->>'unit_price_paise')::integer,
   771→   771        (v_item->>'quantity')::integer,
   772→   772        (v_item->>'line_total_paise')::integer
   773→   773      );
   774→   774    end loop;
   775→   775  
   776→   776    -- Verify computed subtotal matches declared subtotal
   777→   777    if v_computed_subtotal <> p_subtotal_paise then
   778→   778      raise exception 'Subtotal mismatch: computed %, declared %',
   779→   779        v_computed_subtotal, p_subtotal_paise;
   780→   780    end if;
   781→   781  
   782→   782    -- ---------------- REWARD LEDGER (REDEEM) ----------------
   783→   783    if p_reward_points_to_redeem > 0 then
   784→   784      insert into public.reward_transactions (
   785→   785        user_id, order_id, points, type, reason, balance_after
   786→   786      ) values (
   787→   787        p_user_id, v_order_id,
   788→   788        -p_reward_points_to_redeem,
   789→   789        'redeem',
   790→   790        'Redeemed for order ' || v_order_number,
   791→   791        v_new_balance
   792→   792      );
   793→   793    end if;
   794→   794  
   795→   795    return jsonb_build_object(
   796→   796      'success', true,
   797→   797      'order_id', v_order_id,
   798→   798      'order_number', v_order_number
   799→   799    );
   800→   800  end;
   801→   801  $$;

   807→   807  -- ============================================================================
   808→   808  -- RPC: attach_razorpay_order_to_order
   809→   809  -- ----------------------------------------------------------------------------
   810→   810  -- Links a Razorpay order_id to a draft order and transitions the
   811→   811  -- order to 'awaiting_payment' status. Also creates the payments row.
   812→   812  -- Idempotent: if called twice with same Razorpay order_id, returns success
   813→   813  -- without creating duplicates.
   814→   814  -- ============================================================================
   815→   815  create or replace function public.attach_razorpay_order_to_order(
   816→   816    p_order_id uuid,
   817→   817    p_user_id uuid,
   818→   818    p_razorpay_order_id text,
   819→   819    p_amount_paise integer
   820→   820  )
   821→   821  returns jsonb
   822→   822  language plpgsql
   823→   823  security definer
   824→   824  set search_path = public
   825→   825  as $$
   826→   826  declare
   827→   827    v_existing_rzp text;
   828→   828    v_payment_id uuid;
   829→   829  begin
   830→   830    -- Lock the order row
   831→   831    select razorpay_order_id into v_existing_rzp
   832→   832    from public.orders
   833→   833    where id = p_order_id and user_id = p_user_id
   834→   834    for update;
   835→   835  
   836→   836    if not found then
   837→   837      return jsonb_build_object('success', false, 'message', 'Order not found');
   838→   838    end if;
   839→   839  
   840→   840    -- If already attached to a different Razorpay order, reject
   841→   841    if v_existing_rzp is not null and v_existing_rzp <> p_razorpay_order_id then
   842→   842      return jsonb_build_object('success', false, 'message', 'Order already has a different payment intent');
   843→   843    end if;
   844→   844  
   845→   845    -- Update order to awaiting_payment (if not already)
   846→   846    update public.orders
   847→   847    set order_status = case when order_status = 'draft' then 'awaiting_payment' else order_status end,
   848→   848        razorpay_order_id = p_razorpay_order_id
   849→   849    where id = p_order_id;
   850→   850  
   851→   851    -- Insert payment row (idempotent: if exists for this Razorpay order, do nothing)
   852→   852    insert into public.payments (
   853→   853      order_id, user_id,
   854→   854      razorpay_order_id, amount_paise, currency, status
   855→   855    ) values (
   856→   856      p_order_id, p_user_id,
   857→   857      p_razorpay_order_id, p_amount_paise, 'inr', 'pending'
   858→   858    )
   859→   859    on conflict (razorpay_order_id) do nothing
   860→   860    returning id into v_payment_id;
   861→   861  
   862→   862    -- If conflict (existing payment), fetch its id
   863→   863    if v_payment_id is null then
   864→   864      select id into v_payment_id
   865→   865      from public.payments
   866→   866      where razorpay_order_id = p_razorpay_order_id
   867→   867      limit 1;
   868→   868    end if;
   869→   869  
   870→   870    return jsonb_build_object(
   871→   871      'success', true,
   872→   872      'order_id', p_order_id,
   873→   873      'payment_id', v_payment_id
   874→   874    );
   875→   875  end;
   876→   876  $$;

   882→   882  -- ============================================================================
   883→   883  -- RPC: mark_order_succeeded
   884→   884  -- ----------------------------------------------------------------------------
   885→   885  -- Called EITHER from:
   886→   886  --  - /api/checkout/verify-payment (PRIMARY — client verifies signature)
   887→   887  --  - /api/razorpay/webhook (SECONDARY — resilience if client closes browser)
   888→   888  --
   889→   889  -- Atomic operations:
   890→   890  --   1. Insert into processed_webhook_events (idempotency guard)
   891→   891  --   2. Update order: payment_status='succeeded', order_status='confirmed'
   892→   892  --   3. Update payment: status='succeeded', razorpay_payment_id, razorpay_signature, raw_payload
   893→   893  --   4. Award reward points (5 if subtotal > ₹500 AND plantation donation)
   894→   894  --   5. Clear user's cart (delete from cart_items)
   895→   895  --
   896→   896  -- If called twice with same event_id, second call is a no-op (success).
   897→   897  -- ============================================================================
   898→   898  create or replace function public.mark_order_succeeded(
   899→   899    p_order_id uuid,
   900→   900    p_razorpay_payment_id text,
   901→   901    p_razorpay_signature text,
   902→   902    p_webhook_event_id text,
   903→   903    p_event_type text,
   904→   904    p_raw_payload jsonb
   905→   905  )
   906→   906  returns jsonb
   907→   907  language plpgsql
   908→   908  security definer
   909→   909  set search_path = public
   910→   910  as $$
   911→   911  declare
   912→   912    v_order public.orders%rowtype;
   913→   913    v_payment_id uuid;
   914→   914    v_balance integer;
   915→   915    v_new_balance integer;
   916→   916    v_points_to_earn integer := 0;
   917→   917    v_user_id uuid;
   918→   918  begin
   919→   919    -- ---------------- IDEMPOTENCY CHECK ----------------
   920→   920    -- Try to record this event. If it already exists, we've processed it.
   921→   921    begin
   922→   922      insert into public.processed_webhook_events (event_id, event_type, payment_id)
   923→   923      values (p_webhook_event_id, p_event_type, null)
   924→   924      returning payment_id into v_payment_id;
   925→   925    exception when unique_violation then
   926→   926      return jsonb_build_object('success', true, 'message', 'Event already processed', 'idempotent', true);
   927→   927    end;
   928→   928  
   929→   929    -- ---------------- LOAD ORDER (lock) ----------------
   930→   930    select * into v_order
   931→   931    from public.orders
   932→   932    where id = p_order_id
   933→   933    for update;
   934→   934  
   935→   935    if not found then
   936→   936      return jsonb_build_object('success', false, 'message', 'Order not found');
   937→   937    end if;
   938→   938  
   939→   939    v_user_id := v_order.user_id;
   940→   940  
   941→   941    -- Find payment row
   942→   942    select id into v_payment_id
   943→   943    from public.payments
   944→   944    where razorpay_order_id = v_order.razorpay_order_id

   956→   956  
   957→   957      -- Backfill payment_id in processed_webhook_events

   959→   959      set payment_id = v_payment_id
   960→   960      where event_id = p_webhook_event_id;
   961→   961    end if;
   962→   962  
   963→   963    -- If order is already confirmed, skip re-processing (idempotency)
   964→   964    if v_order.order_status = 'confirmed' then
   965→   965      return jsonb_build_object('success', true, 'message', 'Order already confirmed', 'idempotent', true);
   966→   966    end if;
   967→   967  
   968→   968    -- ---------------- UPDATE ORDER STATUS ----------------
   969→   969    update public.orders
   970→   970    set payment_status = 'succeeded',
   971→   971        order_status = 'confirmed'
   972→   972    where id = p_order_id;
   973→   973  
   974→   974    -- ---------------- AWARD REWARD POINTS ----------------
   975→   975    -- Rule: 5 points if BOTH conditions true:
   976→   976    --   1. subtotal_paise > 50000 (₹500)
   977→   977    --   2. donation_plantation_paise = 500 (₹5)
   978→   978    if v_order.subtotal_paise > 50000 and v_order.donation_plantation_paise = 500 then
   979→   979      v_points_to_earn := 5;
   980→   980    end if;
   981→   981  
   982→   982    if v_points_to_earn > 0 then
   983→   983      -- Lazy init balance row
   984→   984      insert into public.reward_balance (user_id)
   985→   985      values (v_user_id)
   986→   986      on conflict (user_id) do nothing;
   987→   987  
   988→   988      select balance_points into v_balance
   989→   989      from public.reward_balance
   990→   990      where user_id = v_user_id
   991→   991      for update;
   992→   992  
   993→   993      v_new_balance := v_balance + v_points_to_earn;
   994→   994  
   995→   995      update public.reward_balance
   996→   996      set balance_points = v_new_balance,
   997→   997          total_earned = total_earned + v_points_to_earn
   998→   998      where user_id = v_user_id;
   999→   999  
  1000→  1000      insert into public.reward_transactions (
  1001→  1001        user_id, order_id, points, type, reason, balance_after
  1002→  1002      ) values (
  1003→  1003        v_user_id, p_order_id,
  1004→  1004        v_points_to_earn,
  1005→  1005        'earn',
  1006→  1006        'Earned from order ' || v_order.order_number,
  1007→  1007        v_new_balance
  1008→  1008      );
  1009→  1009  
  1010→  1010      -- Record earned points on the order
  1011→  1011      update public.orders
  1012→  1012      set reward_points_earned = v_points_to_earn
  1013→  1013      where id = p_order_id;
  1014→  1014    end if;
  1015→  1015  
  1016→  1016    -- ---------------- CLEAR USER'S CART ----------------
  1017→  1017    -- Cart is now an order — empty the plate.
  1018→  1018    delete from public.cart_items where user_id = v_user_id;
  1019→  1019  
  1020→  1020    return jsonb_build_object(
  1021→  1021      'success', true,
  1022→  1022      'order_id', p_order_id,
  1023→  1023      'order_number', v_order.order_number,
  1024→  1024      'reward_points_earned', v_points_to_earn
  1025→  1025    );
  1026→  1026  end;
