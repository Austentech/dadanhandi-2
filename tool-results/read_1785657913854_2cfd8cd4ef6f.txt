   518→-- ============================================================================
   519→-- RPC: preview_reward_redemption
   520→-- ----------------------------------------------------------------------------
   521→-- Returns the discount (paise) the user would get for redeeming N points,
   522→-- WITHOUT actually deducting. Pure calc with validation.
   523→-- Rules: 10 points = ₹5 (500 paise). Must be multiple of 10.
   524→-- ============================================================================
   525→create or replace function public.preview_reward_redemption(
   526→  p_user_id uuid,
   527→  p_points integer
   528→)
   529→returns jsonb
   530→language plpgsql
   531→security definer
   532→set search_path = public
   533→as $$
   534→declare
   535→  v_balance integer;
   536→  v_discount_paise integer;
   537→begin
   538→  if p_points is null or p_points < 0 then
   539→    return jsonb_build_object('success', false, 'message', 'Invalid points value');
   540→  end if;
   541→
   542→  if p_points % 10 <> 0 then
   543→    return jsonb_build_object('success', false, 'message', 'Points must be a multiple of 10');
   544→  end if;
   545→
   546→  if p_points < 10 then
   547→    return jsonb_build_object('success', false, 'message', 'Minimum 10 points required');
   548→  end if;
   549→
   550→  -- Get current balance (lazy init)
   551→  insert into public.reward_balance (user_id)
   552→  values (p_user_id)
   553→  on conflict (user_id) do nothing;
   554→
   555→  select balance_points into v_balance
   556→  from public.reward_balance
   557→  where user_id = p_user_id
   558→  for update;  -- lock row for the duration of this transaction
   559→
   560→  if v_balance is null then
   561→    return jsonb_build_object('success', false, 'message', 'Balance not found');
   562→  end if;
   563→
   564→  if p_points > v_balance then
   565→    return jsonb_build_object(
   566→      'success', false,
   567→      'message', 'Insufficient points',
   568→      'available', v_balance
   569→    );
   570→  end if;
   571→
   572→  -- 10 points = 500 paise → discount = (points / 10) * 500
   573→  v_discount_paise := (p_points / 10) * 500;
   574→
   575→  return jsonb_build_object(
   576→    'success', true,
   577→    'points', p_points,
   578→    'discount_paise', v_discount_paise,
   579→    'balance_after', v_balance - p_points
   580→  );
   581→end;
   582→$$;
   583→
   584→grant execute on function public.preview_reward_redemption(uuid, integer) to authenticated;
   585→
   586→-- ============================================================================
   587→-- RPC: create_draft_order
   588→-- ----------------------------------------------------------------------------
   589→-- Atomic creation of:
   590→--   1. Order row (status='draft', payment_status='pending')
   591→--   2. Order items rows (from p_items JSON array)
   592→--   3. Reward point deduction (if p_reward_points_to_redeem > 0)
   593→--      - Locks reward_balance row
   594→--      - Verifies sufficient balance
   595→--      - Deducts + inserts reward_transactions ledger row
   596→--
   597→-- Idempotency:
   598→--   - p_idempotency_key is UNIQUE. If the same key is submitted twice,
   599→--     the second call fails with a unique constraint violation, which
   600→--     the application layer catches and returns the original order.
   601→--
   602→-- Item JSON shape:
   603→--   [{line_key, item_id, variant_id, item_name, item_emoji, item_type,
   604→--     variant_label, weight_grams, piece_count, unit_price_paise,
   605→--     quantity, line_total_paise}, ...]
   606→--
   607→-- ALL prices are recomputed at the APPLICATION layer (cart-service) before
   608→-- this function is called. The function still enforces:
   609→--   - subtotal = SUM(line_total_paise) for the order's items
   610→--   - final_amount = subtotal + donations - reward_discount
   611→--   - reward_points_to_redeem is multiple of 10 and <= balance
   612→-- ============================================================================
   613→create or replace function public.create_draft_order(
   614→  p_user_id uuid,
   615→  p_branch_id uuid,
   616→  p_pickup_date date,
   617→  p_slot_start time,
   618→  p_slot_end time,
   619→  p_subtotal_paise integer,
   620→  p_donation_plantation_paise integer,
   621→  p_donation_hunger_paise integer,
   622→  p_reward_points_to_redeem integer,
   623→  p_reward_discount_paise integer,
   624→  p_final_amount_paise integer,
   625→  p_idempotency_key text,
   626→  p_items jsonb
   627→)
   628→returns jsonb
   629→language plpgsql
   630→security definer
   631→set search_path = public
   632→as $$
   633→declare
   634→  v_order_id uuid;
   635→  v_order_number text;
   636→  v_balance integer;
   637→  v_new_balance integer;
   638→  v_computed_subtotal integer := 0;
   639→  v_computed_final integer;
   640→  v_item jsonb;
   641→  v_expected_discount integer;
   642→  v_expected_final integer;
   643→begin
   644→  -- ---------------- VALIDATION ----------------
   645→  if p_subtotal_paise < 0 then
   646→    return jsonb_build_object('success', false, 'message', 'Invalid subtotal');
   647→  end if;
   648→
   649→  if p_donation_plantation_paise not in (0, 500) then
   650→    return jsonb_build_object('success', false, 'message', 'Invalid plantation donation');
   651→  end if;
   652→
   653→  if p_donation_hunger_paise not in (0, 1000) then
   654→    return jsonb_build_object('success', false, 'message', 'Invalid hunger donation');
   655→  end if;
   656→
   657→  if p_reward_points_to_redeem < 0 or p_reward_points_to_redeem % 10 <> 0 then
   658→    return jsonb_build_object('success', false, 'message', 'Invalid reward points');
   659→  end if;
   660→
   661→  if p_reward_discount_paise < 0 then
   662→    return jsonb_build_object('success', false, 'message', 'Invalid reward discount');
   663→  end if;
   664→
   665→  if p_final_amount_paise < 0 then
   666→    return jsonb_build_object('success', false, 'message', 'Invalid final amount');
   667→  end if;
   668→
   669→  -- Verify reward discount matches points redeemed
   670→  if p_reward_points_to_redeem > 0 then
   671→    v_expected_discount := (p_reward_points_to_redeem / 10) * 500;
   672→    if p_reward_discount_paise <> v_expected_discount then
   673→      return jsonb_build_object('success', false, 'message', 'Reward discount mismatch');
   674→    end if;
   675→  else
   676→    if p_reward_discount_paise <> 0 then
   677→      return jsonb_build_object('success', false, 'message', 'Reward discount without redemption');
   678→    end if;
   679→  end if;
   680→
   681→  -- Verify final amount = subtotal + donations - reward_discount
   682→  v_expected_final := p_subtotal_paise + p_donation_plantation_paise
   683→                      + p_donation_hunger_paise - p_reward_discount_paise;
   684→  if p_final_amount_paise <> v_expected_final then
   685→    return jsonb_build_object('success', false, 'message', 'Final amount mismatch');
   686→  end if;
   687→
   688→  -- If redeeming points: verify balance + deduct atomically
   689→  if p_reward_points_to_redeem > 0 then
   690→    -- Lazy init balance row
   691→    insert into public.reward_balance (user_id)
   692→    values (p_user_id)
   693→    on conflict (user_id) do nothing;
   694→
   695→    select balance_points into v_balance
   696→    from public.reward_balance
   697→    where user_id = p_user_id
   698→    for update;  -- row lock
   699→
   700→    if v_balance is null then
   701→      return jsonb_build_object('success', false, 'message', 'Reward balance not found');
   702→    end if;
   703→
   704→    if p_reward_points_to_redeem > v_balance then
   705→      return jsonb_build_object(
   706→        'success', false,
   707→        'message', 'Insufficient reward points',
   708→        'available', v_balance
   709→      );
   710→    end if;
   711→
   712→    v_new_balance := v_balance - p_reward_points_to_redeem;
   713→
   714→    update public.reward_balance
   715→    set balance_points = v_new_balance,
   716→        total_redeemed = total_redeemed + p_reward_points_to_redeem
   717→    where user_id = p_user_id;
   718→  end if;
   719→
   720→  -- ---------------- CREATE ORDER ----------------
   721→  v_order_number := public.generate_order_number();
   722→
   723→  insert into public.orders (
   724→    order_number, user_id, branch_id,
   725→    pickup_date, pickup_slot_start, pickup_slot_end,
   726→    subtotal_paise,
   727→    donation_plantation_paise, donation_hunger_paise,
   728→    reward_points_redeemed, reward_discount_paise,
   729→    final_amount_paise,
   730→    payment_status, order_status,
   731→    idempotency_key
   732→  ) values (
   733→    v_order_number, p_user_id, p_branch_id,
   734→    p_pickup_date, p_slot_start, p_slot_end,
   735→    p_subtotal_paise,
   736→    p_donation_plantation_paise, p_donation_hunger_paise,
   737→    p_reward_points_to_redeem, p_reward_discount_paise,
   738→    p_final_amount_paise,
   739→    'pending', 'draft',
   740→    p_idempotency_key
   741→  )
   742→  returning id into v_order_id;
   743→
   744→  -- ---------------- INSERT ITEMS ----------------
   745→  if jsonb_array_length(p_items) = 0 then
   746→    raise exception 'Cannot create order with empty items';
   747→  end if;
   748→
   749→  for v_item in select * from jsonb_array_elements(p_items) loop
   750→    -- Accumulate subtotal for verification
   751→    v_computed_subtotal := v_computed_subtotal + (v_item->>'line_total_paise')::integer;
   752→
   753→    insert into public.order_items (
   754→      order_id, user_id,
   755→      line_key, item_id, variant_id,
   756→      item_name, item_emoji, item_type, variant_label,
   757→      weight_grams, piece_count,
   758→      unit_price_paise, quantity, line_total_paise
   759→    ) values (
   760→      v_order_id, p_user_id,
   761→      v_item->>'line_key',
   762→      v_item->>'item_id',
   763→      v_item->>'variant_id',
   764→      v_item->>'item_name',
   765→      v_item->>'item_emoji',
   766→      v_item->>'item_type',
   767→      v_item->>'variant_label',
   768→      nullif(v_item->>'weight_grams', '')::integer,
   769→      nullif(v_item->>'piece_count', '')::integer,
   770→      (v_item->>'unit_price_paise')::integer,
   771→      (v_item->>'quantity')::integer,
   772→      (v_item->>'line_total_paise')::integer
   773→    );
   774→  end loop;
   775→
   776→  -- Verify computed subtotal matches declared subtotal
   777→  if v_computed_subtotal <> p_subtotal_paise then
   778→    raise exception 'Subtotal mismatch: computed %, declared %',
   779→      v_computed_subtotal, p_subtotal_paise;
   780→  end if;
   781→
   782→  -- ---------------- REWARD LEDGER (REDEEM) ----------------
   783→  if p_reward_points_to_redeem > 0 then
   784→    insert into public.reward_transactions (
   785→      user_id, order_id, points, type, reason, balance_after
   786→    ) values (
   787→      p_user_id, v_order_id,
   788→      -p_reward_points_to_redeem,
   789→      'redeem',
   790→      'Redeemed for order ' || v_order_number,
   791→      v_new_balance
   792→    );
   793→  end if;
   794→
   795→  return jsonb_build_object(
   796→    'success', true,
   797→    'order_id', v_order_id,
   798→    'order_number', v_order_number
   799→  );
   800→end;

   807→-- ============================================================================
   808→-- RPC: attach_razorpay_order_to_order
   809→-- ----------------------------------------------------------------------------
   810→-- Links a Razorpay order_id to a draft order and transitions the
   811→-- order to 'awaiting_payment' status. Also creates the payments row.
   812→-- Idempotent: if called twice with same Razorpay order_id, returns success
   813→-- without creating duplicates.
   814→-- ============================================================================
   815→create or replace function public.attach_razorpay_order_to_order(
   816→  p_order_id uuid,
   817→  p_user_id uuid,
   818→  p_razorpay_order_id text,
   819→  p_amount_paise integer
   820→)
   821→returns jsonb
   822→language plpgsql
   823→security definer
   824→set search_path = public
   825→as $$
   826→declare
   827→  v_existing_rzp text;
   828→  v_payment_id uuid;
   829→begin
   830→  -- Lock the order row
   831→  select razorpay_order_id into v_existing_rzp
   832→  from public.orders
   833→  where id = p_order_id and user_id = p_user_id
   834→  for update;
   835→
   836→  if not found then
   837→    return jsonb_build_object('success', false, 'message', 'Order not found');
   838→  end if;
   839→
   840→  -- If already attached to a different Razorpay order, reject
   841→  if v_existing_rzp is not null and v_existing_rzp <> p_razorpay_order_id then
   842→    return jsonb_build_object('success', false, 'message', 'Order already has a different payment intent');
   843→  end if;
   844→
   845→  -- Update order to awaiting_payment (if not already)
   846→  update public.orders
   847→  set order_status = case when order_status = 'draft' then 'awaiting_payment' else order_status end,
   848→      razorpay_order_id = p_razorpay_order_id
   849→  where id = p_order_id;
   850→

   852→  insert into public.payments (
   853→    order_id, user_id,
   854→    razorpay_order_id, amount_paise, currency, status
   855→  ) values (
   856→    p_order_id, p_user_id,
   857→    p_razorpay_order_id, p_amount_paise, 'inr', 'pending'
   858→  )
   859→  on conflict (razorpay_order_id) do nothing
   860→  returning id into v_payment_id;
   861→
   862→  -- If conflict (existing payment), fetch its id
   863→  if v_payment_id is null then
   864→    select id into v_payment_id
   865→    from public.payments
   866→    where razorpay_order_id = p_razorpay_order_id
   867→    limit 1;
   868→  end if;
   869→
   870→  return jsonb_build_object(
   871→    'success', true,
   872→    'order_id', p_order_id,
   873→    'payment_id', v_payment_id
   874→  );
   875→end;
   876→$$;

   878→grant execute on function public.attach_razorpay_order_to_order(
   879→  uuid, uuid, text, integer
   880→) to authenticated;

   882→-- ============================================================================
   883→-- RPC: mark_order_succeeded
   884→-- ----------------------------------------------------------------------------
   885→-- Called EITHER from:
   886→--  - /api/checkout/verify-payment (PRIMARY — client verifies signature)
   887→--  - /api/razorpay/webhook (SECONDARY — resilience if client closes browser)
   888→--
   889→-- Atomic operations:
   890→--   1. Insert into processed_webhook_events (idempotency guard)
   891→--   2. Update order: payment_status='succeeded', order_status='confirmed'
   892→--   3. Update payment: status='succeeded', razorpay_payment_id, razorpay_signature, raw_payload
   893→--   4. Award reward points (5 if subtotal > ₹500 AND plantation donation)
   894→--   5. Clear user's cart (delete from cart_items)
   895→--
   896→-- If called twice with same event_id, second call is a no-op (success).
   897→-- ============================================================================
   898→create or replace function public.mark_order_succeeded(
   899→  p_order_id uuid,
   900→  p_razorpay_payment_id text,
   901→  p_razorpay_signature text,
   902→  p_webhook_event_id text,
   903→  p_event_type text,
   904→  p_raw_payload jsonb
   905→)
   906→returns jsonb
   907→language plpgsql
   908→security definer
   909→set search_path = public
   910→as $$
   911→declare
   912→  v_order public.orders%rowtype;
   913→  v_payment_id uuid;
   914→  v_balance integer;
   915→  v_new_balance integer;
   916→  v_points_to_earn integer := 0;
   917→  v_user_id uuid;
   918→begin
   919→  -- ---------------- IDEMPOTENCY CHECK ----------------
   920→  -- Try to record this event. If it already exists, we've processed it.
   921→  begin
   922→    insert into public.processed_webhook_events (event_id, event_type, payment_id)
   923→    values (p_webhook_event_id, p_event_type, null)
   924→    returning payment_id into v_payment_id;
   925→  exception when unique_violation then
   926→    return jsonb_build_object('success', true, 'message', 'Event already processed', 'idempotent', true);
   927→  end;
   928→
   929→  -- ---------------- LOAD ORDER (lock) ----------------
   930→  select * into v_order
   931→  from public.orders
   932→  where id = p_order_id
   933→  for update;
   934→
   935→  if not found then
   936→    return jsonb_build_object('success', false, 'message', 'Order not found');
   937→  end if;
   938→
   939→  v_user_id := v_order.user_id;
   940→
   941→  -- Find payment row
   942→  select id into v_payment_id
   943→  from public.payments
   944→  where razorpay_order_id = v_order.razorpay_order_id
   945→  limit 1;
   946→
   947→  -- Update payment record
   948→  if v_payment_id is not null then
   949→    update public.payments
   950→    set status = 'succeeded',
   951→        razorpay_payment_id = p_razorpay_payment_id,
   952→        razorpay_signature = p_razorpay_signature,
   953→        raw_payload = p_raw_payload,
   954→        webhook_processed_at = now()
   955→    where id = v_payment_id;
   956→
   957→    -- Backfill payment_id in processed_webhook_events
   958→    update public.processed_webhook_events
   959→    set payment_id = v_payment_id
   960→    where event_id = p_webhook_event_id;
   961→  end if;
   962→
   963→  -- If order is already confirmed, skip re-processing (idempotency)
   964→  if v_order.order_status = 'confirmed' then
   965→    return jsonb_build_object('success', true, 'message', 'Order already confirmed', 'idempotent', true);
   966→  end if;
   967→
   968→  -- ---------------- UPDATE ORDER STATUS ----------------
   969→  update public.orders
   970→  set payment_status = 'succeeded',
   971→      order_status = 'confirmed'
   972→  where id = p_order_id;
   973→
   974→  -- ---------------- AWARD REWARD POINTS ----------------
   975→  -- Rule: 5 points if BOTH conditions true:
   976→  --   1. subtotal_paise > 50000 (₹500)
   977→  --   2. donation_plantation_paise = 500 (₹5)
   978→  if v_order.subtotal_paise > 50000 and v_order.donation_plantation_paise = 500 then
   979→    v_points_to_earn := 5;
   980→  end if;
   981→
   982→  if v_points_to_earn > 0 then
   983→    -- Lazy init balance row
   984→    insert into public.reward_balance (user_id)
   985→    values (v_user_id)
   986→    on conflict (user_id) do nothing;
   987→
   988→    select balance_points into v_balance
   989→    from public.reward_balance
   990→    where user_id = v_user_id
   991→    for update;
   992→
   993→    v_new_balance := v_balance + v_points_to_earn;
   994→
   995→    update public.reward_balance
   996→    set balance_points = v_new_balance,
   997→        total_earned = total_earned + v_points_to_earn
   998→    where user_id = v_user_id;
   999→
  1000→    insert into public.reward_transactions (
  1001→      user_id, order_id, points, type, reason, balance_after
  1002→    ) values (
  1003→      v_user_id, p_order_id,
  1004→      v_points_to_earn,
  1005→      'earn',
  1006→      'Earned from order ' || v_order.order_number,
  1007→      v_new_balance
  1008→    );
  1009→
  1010→    -- Record earned points on the order
  1011→    update public.orders
  1012→    set reward_points_earned = v_points_to_earn
  1013→    where id = p_order_id;
  1014→  end if;
  1015→
  1016→  -- ---------------- CLEAR USER'S CART ----------------
  1017→  -- Cart is now an order — empty the plate.
  1018→  delete from public.cart_items where user_id = v_user_id;
  1019→
  1020→  return jsonb_build_object(
  1021→    'success', true,
  1022→    'order_id', p_order_id,
  1023→    'order_number', v_order.order_number,
  1024→    'reward_points_earned', v_points_to_earn
  1025→  );
  1026→end;
  1027→$$;
