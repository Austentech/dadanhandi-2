   410→   519→-- RPC: preview_reward_redemption
   411→   520→-- ----------------------------------------------------------------------------
   412→   521→-- Returns the discount (paise) the user would get for redeeming N points,
   413→   522→-- WITHOUT actually deducting. Pure calc with validation.
   414→   523→-- Rules: 10 points = ₹5 (500 paise). Must be multiple of 10.
   415→   524→-- ============================================================================
   416→   525→create or replace function public.preview_reward_redemption(
   417→   526→  p_user_id uuid,
   418→   527→  p_points integer
   419→   528→)
   420→   529→returns jsonb
   421→   530→language plpgsql
   422→   531→security definer
   423→   532→set search_path = public
   424→   533→as $$
   425→   534→declare
   426→   535→  v_balance integer;
   427→   536→  v_discount_paise integer;
   428→   537→begin
   429→   538→  if p_points is null or p_points < 0 then
   430→   539→    return jsonb_build_object('success', false, 'message', 'Invalid points value');
   431→   540→  end if;
   432→   541→
   433→   542→  if p_points % 10 <> 0 then
   434→   543→    return jsonb_build_object('success', false, 'message', 'Points must be a multiple of 10');
   435→   544→  end if;
   436→   545→
   437→   546→  if p_points < 10 then
   438→   547→    return jsonb_build_object('success', false, 'message', 'Minimum 10 points required');
   439→   548→  end if;
   440→   549→
   441→   550→  -- Get current balance (lazy init)
   442→   551→  insert into public.reward_balance (user_id)
   443→   552→  values (p_user_id)
   444→   553→  on conflict (user_id) do nothing;
   445→   554→
   446→   555→  select balance_points into v_balance
   447→   556→  from public.reward_balance
   448→   557→  where user_id = p_user_id
   449→   558→  for update;  -- lock row for the duration of this transaction
   450→   559→
   451→   560→  if v_balance is null then
   452→   561→    return jsonb_build_object('success', false, 'message', 'Balance not found');
   453→   562→  end if;
   454→   563→
   455→   564→  if p_points > v_balance then
   456→   565→    return jsonb_build_object(
   457→   566→      'success', false,
   458→   567→      'message', 'Insufficient points',
   459→   568→      'available', v_balance
   460→   569→    );
   461→   570→  end if;
   462→   571→
   463→   572→  -- 10 points = 500 paise → discount = (points / 10) * 500
   464→   573→  v_discount_paise := (p_points / 10) * 500;
   465→   574→
   466→   575→  return jsonb_build_object(
   467→   576→    'success', true,
   468→   577→    'points', p_points,
   469→   578→    'discount_paise', v_discount_paise,
   470→   579→    'balance_after', v_balance - p_points
   471→   580→  );
   472→   581→end;
   473→   582→$$;
   474→   583→
   475→   584→grant execute on function public.preview_reward_redemption(uuid, integer) to authenticated;

   477→   586→-- ============================================================================
   478→   587→-- RPC: create_draft_order
   479→   588→-- ----------------------------------------------------------------------------
   480→   589→-- Atomic creation of:
   481→   590→--   1. Order row (status='draft', payment_status='pending')
   482→   591→--   2. Order items rows (from p_items JSON array)
   483→   592→--   3. Reward point deduction (if p_reward_points_to_redeem > 0)
   484→   593→--      - Locks reward_balance row
   485→   594→--      - Verifies sufficient balance
   486→   595→--      - Deducts + inserts reward_transactions ledger row
   487→   596→--
   488→   597→-- Idempotency:
   489→   598→--   - p_idempotency_key is UNIQUE. If the same key is submitted twice,
   490→   599→--     the second call fails with a unique constraint violation, which
   491→   600→--     the application layer catches and returns the original order.
   492→   601→--
   493→   602→-- Item JSON shape:
   494→   603→--   [{line_key, item_id, variant_id, item_name, item_emoji, item_type,
   495→   604→--     variant_label, weight_grams, piece_count, unit_price_paise,
   496→   605→--     quantity, line_total_paise}, ...]
   497→   606→--
   498→   607→-- ALL prices are recomputed at the APPLICATION layer (cart-service) before
   499→   608→-- this function is called. The function still enforces:
   500→   609→--   - subtotal = SUM(line_total_paise) for the order's items
   501→   610→--   - final_amount = subtotal + donations - reward_discount
   502→   611→--   - reward_points_to_redeem is multiple of 10 and <= balance
   503→   612→-- ============================================================================
   504→   613→create or replace function public.create_draft_order(
   505→   614→  p_user_id uuid,
   506→   615→  p_branch_id uuid,
   507→   616→  p_pickup_date date,
   508→   617→  p_slot_start time,
   509→   618→  p_slot_end time,
   510→   619→  p_subtotal_paise integer,
   511→   620→  p_donation_plantation_paise integer,
   512→   621→  p_donation_hunger_paise integer,
   513→   622→  p_reward_points_to_redeem integer,
   514→   623→  p_reward_discount_paise integer,
   515→   624→  p_final_amount_paise integer,
   516→   625→  p_idempotency_key text,
   517→   626→  p_items jsonb
   518→   627→)
   519→   628→returns jsonb
   520→   629→language plpgsql
   521→   630→security definer
   522→   631→set search_path = public
   523→   632→as $$
   524→   633→declare
   525→   634→  v_order_id uuid;
   526→   635→  v_order_number text;
   527→   636→  v_balance integer;
   528→   637→  v_new_balance integer;
   529→   638→  v_computed_subtotal integer := 0;
   530→   639→  v_computed_final integer;
   531→   640→  v_item jsonb;
   532→   641→  v_expected_discount integer;
   533→   642→  v_expected_final integer;

   535→   644→  -- ---------------- VALIDATION ----------------
   536→   645→  if p_subtotal_paise < 0 then
   537→   646→    return jsonb_build_object('success', false, 'message', 'Invalid subtotal');
   538→   647→  end if;
   539→   648→
   540→   649→  if p_donation_plantation_paise not in (0, 500) then
   541→   650→    return jsonb_build_object('success', false, 'message', 'Invalid plantation donation');
   542→   651→  end if;
   543→   652→
   544→   653→  if p_donation_hunger_paise not in (0, 1000) then
   545→   654→    return jsonb_build_object('success', false, 'message', 'Invalid hunger donation');
   546→   655→  end if;
   547→   656→
   548→   657→  if p_reward_points_to_redeem < 0 or p_reward_points_to_redeem % 10 <> 0 then
   549→   658→    return jsonb_build_object('success', false, 'message', 'Invalid reward points');
   550→   659→  end if;
   551→   660→
   552→   661→  if p_reward_discount_paise < 0 then
   553→   662→    return jsonb_build_object('success', false, 'message', 'Invalid reward discount');
   554→   663→  end if;
   555→   664→
   556→   665→  if p_final_amount_paise < 0 then
   557→   666→    return jsonb_build_object('success', false, 'message', 'Invalid final amount');
   558→   667→  end if;
   559→   668→
   560→   669→  -- Verify reward discount matches points redeemed
   561→   670→  if p_reward_points_to_redeem > 0 then
   562→   671→    v_expected_discount := (p_reward_points_to_redeem / 10) * 500;
   563→   672→    if p_reward_discount_paise <> v_expected_discount then
   564→   673→      return jsonb_build_object('success', false, 'message', 'Reward discount mismatch');
   565→   674→    end if;
   566→   675→  else
   567→   676→    if p_reward_discount_paise <> 0 then
   568→   677→      return jsonb_build_object('success', false, 'message', 'Reward discount without redemption');
   569→   678→    end if;
   570→   679→  end if;
   571→   680→
   572→   681→  -- Verify final amount = subtotal + donations - reward_discount
   573→   682→  v_expected_final := p_subtotal_paise + p_donation_plantation_paise
   574→   683→                      + p_donation_hunger_paise - p_reward_discount_paise;
   575→   684→  if p_final_amount_paise <> v_expected_final then
   576→   685→    return jsonb_build_object('success', false, 'message', 'Final amount mismatch');
   577→   686→  end if;
   578→   687→
   579→   688→  -- If redeeming points: verify balance + deduct atomically
   580→   689→  if p_reward_points_to_redeem > 0 then
   581→   690→    -- Lazy init balance row
   582→   691→    insert into public.reward_balance (user_id)
   583→   692→    values (p_user_id)
   584→   693→    on conflict (user_id) do nothing;

   587→   696→    from public.reward_balance
   588→   697→    where user_id = p_user_id

   590→   699→

   592→   701→      return jsonb_build_object('success', false, 'message', 'Reward balance not found');
   593→   702→    end if;
   594→   703→
   595→   704→    if p_reward_points_to_redeem > v_balance then
   596→   705→      return jsonb_build_object(

   598→   707→        'message', 'Insufficient reward points',
   599→   708→        'available', v_balance
   600→   709→      );
   601→   710→    end if;
   602→   711→
   603→   712→    v_new_balance := v_balance - p_reward_points_to_redeem;
   604→   713→
   605→   714→    update public.reward_balance
   606→   715→    set balance_points = v_new_balance,
   607→   716→        total_redeemed = total_redeemed + p_reward_points_to_redeem
   608→   717→    where user_id = p_user_id;
   609→   718→  end if;

   615→   724→    order_number, user_id, branch_id,
   616→   725→    pickup_date, pickup_slot_start, pickup_slot_end,
   617→   726→    subtotal_paise,
   618→   727→    donation_plantation_paise, donation_hunger_paise,
   619→   728→    reward_points_redeemed, reward_discount_paise,
   620→   729→    final_amount_paise,
   621→   730→    payment_status, order_status,
   622→   731→    idempotency_key
   623→   732→  ) values (
   624→   733→    v_order_number, p_user_id, p_branch_id,
   625→   734→    p_pickup_date, p_slot_start, p_slot_end,
   626→   735→    p_subtotal_paise,
   627→   736→    p_donation_plantation_paise, p_donation_hunger_paise,
   628→   737→    p_reward_points_to_redeem, p_reward_discount_paise,
   629→   738→    p_final_amount_paise,
   630→   739→    'pending', 'draft',
   631→   740→    p_idempotency_key
   632→   741→  )
   633→   742→  returning id into v_order_id;
   634→   743→
   635→   744→  -- ---------------- INSERT ITEMS ----------------
   636→   745→  if jsonb_array_length(p_items) = 0 then
   637→   746→    raise exception 'Cannot create order with empty items';
   638→   747→  end if;
   639→   748→
   640→   749→  for v_item in select * from jsonb_array_elements(p_items) loop
   641→   750→    -- Accumulate subtotal for verification
   642→   751→    v_computed_subtotal := v_computed_subtotal + (v_item->>'line_total_paise')::integer;
   643→   752→
   644→   753→    insert into public.order_items (
   645→   754→      order_id, user_id,
   646→   755→      line_key, item_id, variant_id,
   647→   756→      item_name, item_emoji, item_type, variant_label,
   648→   757→      weight_grams, piece_count,
   649→   758→      unit_price_paise, quantity, line_total_paise
   650→   759→    ) values (
   651→   760→      v_order_id, p_user_id,
   652→   761→      v_item->>'line_key',
   653→   762→      v_item->>'item_id',
   654→   763→      v_item->>'variant_id',
   655→   764→      v_item->>'item_name',
   656→   765→      v_item->>'item_emoji',
   657→   766→      v_item->>'item_type',
   658→   767→      v_item->>'variant_label',
   659→   768→      nullif(v_item->>'weight_grams', '')::integer,
   660→   769→      nullif(v_item->>'piece_count', '')::integer,
   661→   770→      (v_item->>'unit_price_paise')::integer,
   662→   771→      (v_item->>'quantity')::integer,
   663→   772→      (v_item->>'line_total_paise')::integer
   664→   773→    );
   665→   774→  end loop;
   666→   775→
   667→   776→  -- Verify computed subtotal matches declared subtotal
   668→   777→  if v_computed_subtotal <> p_subtotal_paise then
   669→   778→    raise exception 'Subtotal mismatch: computed %, declared %',
   670→   779→      v_computed_subtotal, p_subtotal_paise;
   671→   780→  end if;
   672→   781→
   673→   782→  -- ---------------- REWARD LEDGER (REDEEM) ----------------
   674→   783→  if p_reward_points_to_redeem > 0 then
   675→   784→    insert into public.reward_transactions (
   676→   785→      user_id, order_id, points, type, reason, balance_after
   677→   786→    ) values (
   678→   787→      p_user_id, v_order_id,
   679→   788→      -p_reward_points_to_redeem,
   680→   789→      'redeem',
   681→   790→      'Redeemed for order ' || v_order_number,
   682→   791→      v_new_balance
   683→   792→    );
   684→   793→  end if;
   685→   794→
   686→   795→  return jsonb_build_object(
   687→   796→    'success', true,
   688→   797→    'order_id', v_order_id,
   689→   798→    'order_number', v_order_number
   690→   799→  );
   691→   800→end;

   698→   807→-- ============================================================================
   699→   808→-- RPC: attach_razorpay_order_to_order
   700→   809→-- ----------------------------------------------------------------------------
   701→   810→-- Links a Razorpay order_id to a draft order and transitions the
   702→   811→-- order to 'awaiting_payment' status. Also creates the payments row.
   703→   812→-- Idempotent: if called twice with same Razorpay order_id, returns success
   704→   813→-- without creating duplicates.
   705→   814→-- ============================================================================
   706→   815→create or replace function public.attach_razorpay_order_to_order(
   707→   816→  p_order_id uuid,
   708→   817→  p_user_id uuid,
   709→   818→  p_razorpay_order_id text,
   710→   819→  p_amount_paise integer
   711→   820→)

   714→   823→security definer

   716→   825→as $$
   717→   826→declare
   718→   827→  v_existing_rzp text;
   719→   828→  v_payment_id uuid;
   720→   829→begin
   721→   830→  -- Lock the order row
   722→   831→  select razorpay_order_id into v_existing_rzp
   723→   832→  from public.orders
   724→   833→  where id = p_order_id and user_id = p_user_id
   725→   834→  for update;
   726→   835→
   727→   836→  if not found then
   728→   837→    return jsonb_build_object('success', false, 'message', 'Order not found');
   729→   838→  end if;
   730→   839→
   731→   840→  -- If already attached to a different Razorpay order, reject
   732→   841→  if v_existing_rzp is not null and v_existing_rzp <> p_razorpay_order_id then
   733→   842→    return jsonb_build_object('success', false, 'message', 'Order already has a different payment intent');
   734→   843→  end if;
   735→   844→
   736→   845→  -- Update order to awaiting_payment (if not already)
   737→   846→  update public.orders
   738→   847→  set order_status = case when order_status = 'draft' then 'awaiting_payment' else order_status end,
   739→   848→      razorpay_order_id = p_razorpay_order_id
   740→   849→  where id = p_order_id;
   741→   850→
   742→   851→  -- Insert payment row (idempotent: if exists for this Razorpay order, do nothing)
   743→   852→  insert into public.payments (
   744→   853→    order_id, user_id,
   745→   854→    razorpay_order_id, amount_paise, currency, status
   746→   855→  ) values (
   747→   856→    p_order_id, p_user_id,
   748→   857→    p_razorpay_order_id, p_amount_paise, 'inr', 'pending'
   749→   858→  )
   750→   859→  on conflict (razorpay_order_id) do nothing
   751→   860→  returning id into v_payment_id;
   752→   861→
   753→   862→  -- If conflict (existing payment), fetch its id
   754→   863→  if v_payment_id is null then
   755→   864→    select id into v_payment_id
   756→   865→    from public.payments
   757→   866→    where razorpay_order_id = p_razorpay_order_id
   758→   867→    limit 1;
   759→   868→  end if;
   760→   869→
   761→   870→  return jsonb_build_object(
   762→   871→    'success', true,
   763→   872→    'order_id', p_order_id,
   764→   873→    'payment_id', v_payment_id
   765→   874→  );
   766→   875→end;
   767→   876→$$;

   773→   882→-- ============================================================================
   774→   883→-- RPC: mark_order_succeeded
   775→   884→-- ----------------------------------------------------------------------------
   776→   885→-- Called EITHER from:
   777→   886→--  - /api/checkout/verify-payment (PRIMARY — client verifies signature)
   778→   887→--  - /api/razorpay/webhook (SECONDARY — resilience if client closes browser)

   785→   894→--   5. Clear user's cart (delete from cart_items)
   786→   895→--
   787→   896→-- If called twice with same event_id, second call is a no-op (success).
   788→   897→-- ============================================================================

   790→   899→  p_order_id uuid,
   791→   900→  p_razorpay_payment_id text,
   792→   901→  p_razorpay_signature text,
   793→   902→  p_webhook_event_id text,
   794→   903→  p_event_type text,
   795→   904→  p_raw_payload jsonb
   796→   905→)

   798→   907→language plpgsql

   802→   911→declare
   803→   912→  v_order public.orders%rowtype;
   804→   913→  v_payment_id uuid;
   805→   914→  v_balance integer;
   806→   915→  v_new_balance integer;
   807→   916→  v_points_to_earn integer := 0;
   808→   917→  v_user_id uuid;

   812→   921→  begin
   813→   922→    insert into public.processed_webhook_events (event_id, event_type, payment_id)
   814→   923→    values (p_webhook_event_id, p_event_type, null)

   818→   927→  end;

   825→   934→

   827→   936→    return jsonb_build_object('success', false, 'message', 'Order not found');
   828→   937→  end if;

   832→   941→  -- Find payment row
   833→   942→  select id into v_payment_id

   835→   944→  where razorpay_order_id = v_order.razorpay_order_id

   851→   960→    where event_id = p_webhook_event_id;
   852→   961→  end if;

   859→   968→  -- ---------------- UPDATE ORDER STATUS ----------------

   868→   977→  --   2. donation_plantation_paise = 500 (₹5)

   892→  1001→      user_id, order_id, points, type, reason, balance_after
   893→  1002→    ) values (
   894→  1003→      v_user_id, p_order_id,
   895→  1004→      v_points_to_earn,
   896→  1005→      'earn',
   897→  1006→      'Earned from order ' || v_order.order_number,
   898→  1007→      v_new_balance
   899→  1008→    );
   900→  1009→
   901→  1010→    -- Record earned points on the order
   902→  1011→    update public.orders
   903→  1012→    set reward_points_earned = v_points_to_earn
   904→  1013→    where id = p_order_id;

   913→  1022→    'order_id', p_order_id,
   914→  1023→    'order_number', v_order.order_number,
   915→  1024→    'reward_points_earned', v_points_to_earn

   917→  1026→end;

   959→  1068→
   960→  1069→  -- ---------------- LOAD ORDER (lock) ----------------
   961→  1070→  select * into v_order
   962→  1071→  from public.orders
   963→  1072→  where id = p_order_id
   964→  1073→  for update;
   965→  1074→
   966→  1075→  if not found then
   967→  1076→    return jsonb_build_object('success', false, 'message', 'Order not found');

   981→  1090→        failure_reason = p_failure_reason,

   989→  1098→  end if;
   990→  1099→
   991→  1100→  -- If order is already failed/confirmed, skip
   992→  1101→  if v_order.order_status in ('failed', 'confirmed') then
   993→  1102→    return jsonb_build_object('success', true, 'message', 'Order already in terminal state', 'idempotent', true);
   994→  1103→  end if;
   995→  1104→
   996→  1105→  -- ---------------- UPDATE ORDER STATUS ----------------
   997→  1106→  update public.orders
   998→  1107→  set payment_status = 'failed',
   999→  1108→      order_status = 'failed'
  1000→  1109→  where id = p_order_id;
  1001→  1110→
  1002→  1111→  -- ---------------- RESTORE REDEEMED POINTS ----------------
  1003→  1112→  if v_order.reward_points_redeemed > 0 then
  1004→  1113→    select balance_points into v_balance
  1005→  1114→    from public.reward_balance
  1006→  1115→    where user_id = v_user_id
  1007→  1116→    for update;
  1008→  1117→
  1009→  1118→    v_new_balance := v_balance + v_order.reward_points_redeemed;
  1010→  1119→
  1011→  1120→    update public.reward_balance
  1012→  1121→    set balance_points = v_new_balance,
  1013→  1122→        total_redeemed = greatest(total_redeemed - v_order.reward_points_redeemed, 0)
  1014→  1123→    where user_id = v_user_id;
  1015→  1124→
  1016→  1125→    insert into public.reward_transactions (
  1017→  1126→      user_id, order_id, points, type, reason, balance_after

  1020→  1129→      v_order.reward_points_redeemed,
  1021→  1130→      'restore',
  1022→  1131→      'Restored from failed order ' || v_order.order_number,
  1023→  1132→      v_new_balance
  1024→  1133→    );
  1025→  1134→  end if;
  1026→  1135→
  1027→  1136→  return jsonb_build_object(
  1028→  1137→    'success', true,
  1029→  1138→    'order_id', p_order_id,
  1030→  1139→    'order_number', v_order.order_number,
  1031→  1140→    'reward_points_restored', v_order.reward_points_redeemed
  1032→  1141→  );
  1033→  1142→end;
  1034→  1143→$$;
