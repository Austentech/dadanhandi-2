    51→/**
    52→ * Fetch dashboard statistics from the database.
    53→ * Returns counts for each order status category.
    54→ *
    55→ * IMPORTANT: "Pending" and "Upcoming" now use the preparation window.
    56→ *   - Pending (Current Queue): confirmed + paid + pickup slot within window
    57→ *   - Upcoming: confirmed + paid + today + pickup slot AFTER window
    58→ */
    59→export async function getDashboardStats(): Promise<DashboardStatsResult> {
    60→  try {
    61→    const supabase = createAdminClient()
    62→    const bounds = getPreparationWindowBounds()
    63→
    64→    // Get today's date range in IST
    65→    const now = new Date()
    66→    const istOffset = 5.5 * 60 * 60 * 1000
    67→    const istTime = new Date(now.getTime() + istOffset)
    68→    const todayStart = new Date(istTime.getFullYear(), istTime.getMonth(), istTime.getDate())
    69→    const todayStartUTC = new Date(todayStart.getTime() - istOffset)
    70→
    71→    const [todayRes, acceptedRes, preparingRes, readyRes, completedRes, cancelledRes] =
    72→      await Promise.all([
    73→        // Today's orders (all non-draft statuses)
    74→        supabase.from('orders').select('id', { count: 'exact', head: true })
    75→          .neq('order_status', 'draft')
    76→          .gte('created_at', todayStartUTC.toISOString()),
    77→        // Accepted
    78→        supabase.from('orders').select('id', { count: 'exact', head: true })
    79→          .eq('order_status', 'accepted'),
    80→        // Preparing
    81→        supabase.from('orders').select('id', { count: 'exact', head: true })
    82→          .eq('order_status', 'preparing'),
    83→        // Ready for pickup
    84→        supabase.from('orders').select('id', { count: 'exact', head: true })
    85→          .eq('order_status', 'ready_for_pickup'),
    86→        // Completed (today)
    87→        supabase.from('orders').select('id', { count: 'exact', head: true })
    88→          .eq('order_status', 'completed')
    89→          .gte('created_at', todayStartUTC.toISOString()),
    90→        // Cancelled (today)
    91→        supabase.from('orders').select('id', { count: 'exact', head: true })
    92→          .eq('order_status', 'cancelled')
    93→          .gte('created_at', todayStartUTC.toISOString()),
    94→      ])
    95→
    96→    // --- Pending (Current Queue): confirmed + paid + in preparation window + overdue ---
    97→    // Includes: orders in window + overdue (slot passed) + past-date unaccepted
    98→    // eslint-disable-next-line @typescript-eslint/no-explicit-any
    99→    const sb = supabase as any
   100→
   101→    const windowEndStr = `${String(Math.floor(bounds.windowEndMinutes / 60)).padStart(2, '0')}:${String(bounds.windowEndMinutes % 60).padStart(2, '0')}:00`
   102→
   103→    // Pending today: in-window + overdue (slot <= windowEnd, no lower bound)
   104→    const { count: pendingTodayCount } = await sb
   105→      .from('orders')
   106→      .select('id', { count: 'exact', head: true })
   107→      .eq('order_status', 'confirmed')
   108→      .eq('payment_status', 'succeeded')
   109→      .eq('pickup_date', bounds.todayDateStr)
   110→      .lte('pickup_slot_start', windowEndStr)
   111→
   112→    // Pending past: overdue from yesterday or earlier
   113→    const { count: pendingPastCount } = await sb
   114→      .from('orders')
   115→      .select('id', { count: 'exact', head: true })
   116→      .eq('order_status', 'confirmed')
   117→      .eq('payment_status', 'succeeded')
   118→      .lt('pickup_date', bounds.todayDateStr)
   119→
   120→    const pendingCount = (pendingTodayCount || 0) + (pendingPastCount || 0)
   121→
   122→    // --- Upcoming: confirmed + paid + today + AFTER preparation window ---
   123→    const { count: upcomingCount } = await sb
   124→      .from('orders')
   125→      .select('id', { count: 'exact', head: true })
   126→      .eq('order_status', 'confirmed')
   127→      .eq('payment_status', 'succeeded')
   128→      .eq('pickup_date', bounds.todayDateStr)
   129→      .gt('pickup_slot_start', windowEndStr)
   130→
   131→    return {
   132→      success: true,
   133→      stats: {
   134→        todayOrders: todayRes.count || 0,
   135→        pendingOrders: pendingCount || 0,
   136→        acceptedOrders: acceptedRes.count || 0,
   137→        preparingOrders: preparingRes.count || 0,
   138→        readyOrders: readyRes.count || 0,
   139→        completedOrders: completedRes.count || 0,
   140→        cancelledOrders: cancelledRes.count || 0,
   141→        upcomingOrders: upcomingCount || 0,
   142→      },
   143→    }
   144→  } catch (err) {
   145→    console.error('[ADMIN ORDER SERVICE] getDashboardStats error:', err)
   146→    return { success: false, stats: emptyStats() }
   147→  }
   148→}

   339→/**
   340→ * List orders with a given status, including items and customer info.
   341→ * For 'confirmed' status:
   342→ *   - Only shows PAID orders (payment_status='succeeded')
   343→ *   - Shows orders in the preparation window (current queue)
   344→ *   - Shows OVERDUE orders whose pickup slot has already passed (never accepted)
   345→ *   - Shows past-date orders (yesterday or earlier, still confirmed + paid)
   346→ *   - Orders with FUTURE pickup slots are HIDDEN until their window begins
   347→ *   - Each order includes an `isOverdue` flag computed server-side
   348→ */
   349→export async function listOrdersByStatus(
   350→  status: AdminOrderStatus,
   351→  options?: {
   352→    branchSlug?: string
   353→    search?: string
   354→    limit?: number
   355→    offset?: number
   356→  }
   357→): Promise<ListOrdersResult> {
   358→  try {
   359→    const supabase = createAdminClient()
   360→    const limit = options?.limit || 50
   361→    const offset = options?.offset || 0
   362→
   363→    // eslint-disable-next-line @typescript-eslint/no-explicit-any
   364→    const sb = supabase as any
   365→
   366→    const selectFields = `
   367→      *,
   368→      order_items (*),
   369→      branches (name, slug, city)
   370→    `
   371→
   372→    let orders: RawOrderRow[]
   373→    let totalCount: number
   374→    // Store window bounds for isOverdue computation (only for confirmed)
   375→    let windowStartMinutes: number | null = null
   376→    let todayDateStr: string | null = null
   377→
   378→    if (status === 'confirmed') {
   379→      // --- Confirmed orders: two-query approach ---
   380→      // 1. Today's orders: in-window + overdue (no lower bound on slot)
   381→      // 2. Past-date overdue: pickup_date < today
   382→      // Future orders (slot after window end) are excluded.
   383→
   384→      const bounds = getPreparationWindowBounds()
   385→      windowStartMinutes = bounds.windowStartMinutes
   386→      todayDateStr = bounds.todayDateStr
   387→
   388→      const windowEndStr = `${String(Math.floor(bounds.windowEndMinutes / 60)).padStart(2, '0')}:${String(bounds.windowEndMinutes % 60).padStart(2, '0')}:00`
   389→
   390→      // Query 1: Today — in-window + overdue (slot <= windowEnd, no .gte lower bound)
   391→      let todayQ = sb
   392→        .from('orders')
   393→        .select(selectFields, { count: 'exact' })
   394→        .eq('order_status', 'confirmed')
   395→        .eq('payment_status', 'succeeded')
   396→        .eq('pickup_date', bounds.todayDateStr)
   397→        .lte('pickup_slot_start', windowEndStr)
   398→        .order('pickup_slot_start', { ascending: true })
   399→
   400→      // Query 2: Past-date overdue (yesterday or earlier, never accepted)
   401→      let pastQ = sb
   402→        .from('orders')
   403→        .select(selectFields)
   404→        .eq('order_status', 'confirmed')
   405→        .eq('payment_status', 'succeeded')
   406→        .lt('pickup_date', bounds.todayDateStr)
   407→        .order('pickup_date', { ascending: false })
   408→        .order('pickup_slot_start', { ascending: false })
   409→        .limit(50) // Safety limit for very old overdue orders
   410→
   411→      // Apply optional branch filter to both queries
   412→      if (options?.branchSlug) {
   413→        todayQ = todayQ.eq('branch_id', options.branchSlug)
   414→        pastQ = pastQ.eq('branch_id', options.branchSlug)
   415→      }
   416→
   417→      const [todayRes, pastRes] = await Promise.all([todayQ, pastQ])
   418→
   419→      if (todayRes.error) {
   420→        console.error('[ADMIN ORDER SERVICE] listOrdersByStatus today query error:', todayRes.error)
   421→        return { success: false, orders: [], totalCount: 0, error: 'Failed to fetch orders' }
   422→      }
   423→
   424→      const todayOrders = (todayRes.data || []) as RawOrderRow[]
   425→      const pastOrders = (pastRes.data || []) as RawOrderRow[]
   426→
   427→      // Merge: past overdue first (most recent past first), then today's
   428→      orders = [...pastOrders, ...todayOrders]
   429→      totalCount = (todayRes.count || 0) + pastOrders.length
   430→    } else {
   431→      // --- Non-confirmed: simple single query ---
   432→      let query = sb
   433→        .from('orders')
   434→        .select(selectFields, { count: 'exact' })
   435→        .eq('order_status', status)
   436→
   437→      query = query
   438→        .order('pickup_slot_start', { ascending: true })
   439→        .range(offset, offset + limit - 1)
   440→
   441→      if (options?.branchSlug) {
   442→        query = query.eq('branch_id', options.branchSlug)
   443→      }
   444→
   445→      const { data, error, count } = await query
   446→
   447→      if (error) {
   448→        console.error('[ADMIN ORDER SERVICE] listOrdersByStatus error:', error)
   449→        return { success: false, orders: [], totalCount: 0, error: 'Failed to fetch orders' }
   450→      }
   451→
   452→      orders = (data || []) as RawOrderRow[]
   453→      totalCount = count || 0
   454→    }
   455→
   456→    // Apply pagination (for confirmed, we already have all orders merged)
   457→    if (status === 'confirmed') {
   458→      orders = orders.slice(offset, offset + limit)
   459→    }
   460→
   461→    if (!orders || orders.length === 0) {
   462→      return { success: true, orders: [], totalCount: 0 }
   463→    }
   464→
   465→    // Fetch customer profiles for all orders
   466→    const userIds = [...new Set(orders.map((o: RawOrderRow) => o.user_id))]
   467→    const { data: profiles } = await supabase
   468→      .from('profiles')
   469→      .select('id, full_name, whatsapp_number, mobile_number')
   470→      .in('id', userIds)
   471→
   472→    const profileMap = new Map<string, { name: string; whatsappNumber: string | null; mobileNumber: string | null }>()
   473→    for (const p of (profiles || []) as RawProfileRow[]) {
   474→      profileMap.set(p.id, {
   475→        name: p.full_name || 'Customer',
   476→        whatsappNumber: p.whatsapp_number || null,
   477→        mobileNumber: p.mobile_number || null,
   478→      })
   479→    }
   480→
   481→    // Search filter (applied in-memory since Supabase free tier doesn't support full-text across joins)
   482→    let filteredOrders = orders as RawOrderRow[]
   483→    if (options?.search) {
   484→      const q = options.search.toLowerCase()
   485→      filteredOrders = filteredOrders.filter((o) => {
   486→        const orderNum = (o.order_number || '').toLowerCase()
   487→        const profile = profileMap.get(o.user_id)
   488→        const customerName = (profile?.name || '').toLowerCase()
   489→        return orderNum.includes(q) || customerName.includes(q)
   490→      })
   491→    }
   492→
   493→    // Helper: compute isOverdue for a confirmed order (server-side, using window bounds)
   494→    const computeIsOverdue = (raw: RawOrderRow): boolean => {
   495→      if (windowStartMinutes === null || todayDateStr === null) return false
   496→      // Past-date orders are always overdue
   497→      if (raw.pickup_date < todayDateStr) return true
   498→      // Today's orders: overdue if slot started before the window
   499→      const [h, m] = raw.pickup_slot_start.split(':').map(Number)
   500→      if (isNaN(h) || isNaN(m)) return false
   501→      const slotMinutes = h * 60 + m
   502→      return slotMinutes < windowStartMinutes
   503→    }
   504→
   505→    // Map to typed response with proper snake_case -> camelCase conversion
   506→    const mapped: AdminOrderWithItems[] = filteredOrders.map((o) => {
   507→      const profile = profileMap.get(o.user_id)
   508→      // Supabase returns the joined table as singular 'branchs' (from 'branches')
   509→      const branch = o.branchs as { name: string; slug: string; city: string } | null
   510→      return {
   511→        ...mapOrderRow(o),
   512→        items: (o.order_items || []).map(mapOrderItemRow),
   513→        branch: branch ? { name: branch.name, slug: branch.slug, city: branch.city } : null,
   514→        customer: profile || { name: 'Customer', whatsappNumber: null, mobileNumber: null },
   515→        isOverdue: status === 'confirmed' ? computeIsOverdue(o) : undefined,
   516→      }
   517→    })
   518→
   519→    return {
   520→      success: true,
   521→      orders: mapped,
   522→      totalCount,
   523→    }
   524→  } catch (err) {
   525→    console.error('[ADMIN ORDER SERVICE] listOrdersByStatus error:', err)
   526→    return { success: false, orders: [], totalCount: 0, error: 'Failed to fetch orders' }
   527→  }
   528→}

   541→/**
   542→ * Accept an order: transition from 'confirmed' to 'accepted'.
   543→ * - Validates order exists, is 'confirmed', and payment succeeded
   544→ * - Prevents duplicate acceptance (idempotent by design)
   545→ * - Records status change in order_status_history
   546→ */
   547→export async function acceptOrder(
   548→  orderId: string,
   549→  adminUserId: string
   550→): Promise<AcceptOrderResult> {
   551→  try {
   552→    const supabase = createAdminClient()
   553→
   554→    // 1. Validate input
   555→    if (!orderId || typeof orderId !== 'string') {
   556→      return { success: false, message: 'Invalid order ID' }
   557→    }
   558→
   559→    // 2. Fetch the order
   560→    // eslint-disable-next-line @typescript-eslint/no-explicit-any
   561→    const { data: order, error: fetchErr }: { data: RawOrderRow | null; error: any } = await supabase
   562→      .from('orders')
   563→      .select('*')
   564→      .eq('id', orderId)
   565→      .single()
   566→
   567→    if (fetchErr || !order) {
   568→      return { success: false, message: 'Order not found' }
   569→    }
   570→
   571→    // 3. Validate payment status — must be succeeded
   572→    if (order.payment_status !== 'succeeded') {
   573→      return { success: false, message: 'Cannot accept an unpaid order' }
   574→    }
   575→
   576→    // 4. Validate current status
   577→    if (order.order_status !== 'confirmed') {
   578→      if (order.order_status === 'accepted') {
   579→        return { success: true, message: 'Order already accepted' }
   580→      }
   581→      return { success: false, message: `Cannot accept order in '${order.order_status}' status` }
   582→    }
   583→
   584→    // 5. Validate the transition is allowed
   585→    const allowedTransitions = VALID_TRANSITIONS[order.order_status] || []
   586→    if (!allowedTransitions.includes('accepted')) {
   587→      return { success: false, message: 'Invalid status transition' }
   588→    }
   589→
   590→    // 6. Update order status (race condition guard: only update if still confirmed)
   591→    // eslint-disable-next-line @typescript-eslint/no-explicit-any
   592→    const sb = supabase as any
   593→    const { error: updateErr } = await sb
   594→      .from('orders')
   595→      .update({
   596→        order_status: 'accepted',
   597→        updated_at: new Date().toISOString(),
   598→      })
   599→      .eq('id', orderId)
   600→      .eq('order_status', 'confirmed')
   601→
   602→    if (updateErr) {
   603→      console.error('[ADMIN ORDER SERVICE] acceptOrder update error:', updateErr)
   604→      return { success: false, message: 'Failed to update order. Please try again.' }
   605→    }
   606→
   607→    // 7. Log status change in history
   608→    //    Table schema: (order_id, status, note, created_at)
   609→    //    Unique constraint: (order_id, status) — one row per status per order
   610→    try {
   611→      const { error: histErr } = await sb.from('order_status_history').insert({
   612→        order_id: orderId,
   613→        status: 'accepted',
   614→        note: 'Accepted by admin',
   615→      })
   616→      // Ignore duplicate key errors (order already has 'accepted' status row)
   617→      if (histErr && !String(histErr.message || histErr).includes('duplicate')) {
   618→        console.error('[ADMIN ORDER SERVICE] Failed to log status history:', histErr)
   619→      }
   620→    } catch (historyErr) {
   621→      // Log but don't fail — the order status was already updated
   622→      console.error('[ADMIN ORDER SERVICE] Failed to log status history:', historyErr)
   623→    }
   624→
   625→    return { success: true, message: 'Order accepted successfully' }
   626→  } catch (err) {
   627→    console.error('[ADMIN ORDER SERVICE] acceptOrder error:', err)
   628→    return { success: false, message: 'Something went wrong. Please try again.' }
   629→  }
   630→}

   636→/**
   637→ * List all ongoing orders (accepted, preparing, ready_for_pickup).
   638→ * Includes items, customer info, branch info, and accepted-at timestamp.
   639→ * Ordered by pickup time (soonest first), then by status priority.
   640→ */
   641→export async function listOngoingOrders(options?: {
   642→  branchSlug?: string
   643→  search?: string
   644→  limit?: number
   645→  offset?: number
   646→}): Promise<ListOrdersResult> {
   647→  try {
   648→    const supabase = createAdminClient()
   649→    const limit = options?.limit || 50
   650→    const offset = options?.offset || 0
   651→
   652→    // eslint-disable-next-line @typescript-eslint/no-explicit-any
   653→    const sb = supabase as any
   654→
   655→    const selectFields = `
   656→      *,
   657→      order_items (*),
   658→      branches (name, slug, city)
   659→    `
   660→
   661→    let query = sb
   662→      .from('orders')
   663→      .select(selectFields, { count: 'exact' })
   664→      .in('order_status', ['accepted', 'preparing', 'ready_for_pickup'])
   665→      .order('pickup_slot_start', { ascending: true })
   666→      .range(offset, offset + limit - 1)
   667→
   668→    if (options?.branchSlug) {
   669→      query = query.eq('branch_id', options.branchSlug)
   670→    }
   671→
   672→    const { data: orders, error, count } = await query
   673→
   674→    if (error) {
   675→      console.error('[ADMIN ORDER SERVICE] listOngoingOrders error:', error)
   676→      return { success: false, orders: [], totalCount: 0, error: 'Failed to fetch ongoing orders' }
   677→    }
   678→
   679→    if (!orders || orders.length === 0) {
   680→      return { success: true, orders: [], totalCount: 0 }
   681→    }
   682→
   683→    const rawOrders = orders as RawOrderRow[]
   684→
   685→    // --- Batch fetch customer profiles ---
   686→    const userIds = [...new Set(rawOrders.map((o) => o.user_id))]
   687→    const { data: profiles } = await supabase
   688→      .from('profiles')
   689→      .select('id, full_name, whatsapp_number, mobile_number')
   690→      .in('id', userIds)
   691→
   692→    const profileMap = new Map<string, { name: string; whatsappNumber: string | null; mobileNumber: string | null }>()
   693→    for (const p of (profiles || []) as RawProfileRow[]) {
   694→      profileMap.set(p.id, {
   695→        name: p.full_name || 'Customer',
   696→        whatsappNumber: p.whatsapp_number || null,
   697→        mobileNumber: p.mobile_number || null,
   698→      })
   699→    }
   700→
   701→    // --- Batch fetch accepted-at timestamps from status history ---
   702→    const orderIds = rawOrders.map((o) => o.id)
   703→    const { data: historyRows } = await supabase
   704→      .from('order_status_history')
   705→      .select('order_id, created_at')
   706→      .in('order_id', orderIds)
   707→      .eq('status', 'accepted')
   708→
   709→    const acceptedAtMap = new Map<string, string>()
   710→    for (const row of (historyRows || []) as { order_id: string; created_at: string }[]) {
   711→      if (!acceptedAtMap.has(row.order_id)) {
   712→        acceptedAtMap.set(row.order_id, row.created_at)
   713→      }
   714→    }
   715→
   716→    // --- Search filter (in-memory) ---
   717→    let filteredOrders = rawOrders
   718→    if (options?.search) {
   719→      const q = options.search.toLowerCase()
   720→      filteredOrders = filteredOrders.filter((o) => {
   721→        const orderNum = (o.order_number || '').toLowerCase()
   722→        const profile = profileMap.get(o.user_id)
   723→        const customerName = (profile?.name || '').toLowerCase()
   724→        return orderNum.includes(q) || customerName.includes(q)
   725→      })
   726→    }
   727→
   728→    // --- Status priority for sorting (accepted=1, preparing=2, ready=3) ---
   729→    const statusWeight: Record<string, number> = {
   730→      accepted: 1,
   731→      preparing: 2,
   732→      ready_for_pickup: 3,
   733→    }
   734→
   735→    // --- Map to typed response ---
   736→    const mapped: AdminOrderWithItems[] = filteredOrders.map((o) => {
   737→      const profile = profileMap.get(o.user_id)
   738→      const branch = o.branchs as { name: string; slug: string; city: string } | null
   739→      return {
   740→        ...mapOrderRow(o),
   741→        items: (o.order_items || []).map(mapOrderItemRow),
   742→        branch: branch ? { name: branch.name, slug: branch.slug, city: branch.city } : null,
   743→        customer: profile || { name: 'Customer', whatsappNumber: null, mobileNumber: null },
   744→        acceptedAt: acceptedAtMap.get(o.id) || undefined,
   745→      }
   746→    })
   747→
   748→    // Sort: pickup time ascending, then status weight ascending (accepted before preparing)
   749→    mapped.sort((a, b) => {
   750→      const slotA = a.pickupSlotStart
   751→      const slotB = b.pickupSlotStart
   752→      if (slotA !== slotB) return slotA.localeCompare(slotB)
   753→      return (statusWeight[a.orderStatus] || 99) - (statusWeight[b.orderStatus] || 99)
   754→    })
   755→
   756→    return {
   757→      success: true,
   758→      orders: mapped,
   759→      totalCount: count || 0,
   760→    }
   761→  } catch (err) {
   762→    console.error('[ADMIN ORDER SERVICE] listOngoingOrders error:', err)
   763→    return { success: false, orders: [], totalCount: 0, error: 'Failed to fetch ongoing orders' }
   764→  }
   765→}

   767→// ============================================================================
   768→// UPDATE ORDER STATUS (Kitchen Workflow)
   769→// ============================================================================
   770→
   771→export interface UpdateOrderStatusResult {
   772→  success: boolean
   773→  message: string
   774→  error?: string
   775→}
   776→
   777→/**
   778→ * Subset of VALID_TRANSITIONS for the kitchen workflow.
   779→ * This module only handles forward transitions:
   780→ *   accepted → preparing → ready_for_pickup
   781→ * Cancellation and other transitions are handled by other modules.
   782→ */
   783→const KITCHEN_TRANSITIONS: Record<string, AdminOrderStatus[]> = {
   784→  accepted: ['preparing'],
   785→  preparing: ['ready_for_pickup'],
   786→}
   787→
   788→/**
   789→ * Update an order's status as part of the kitchen workflow.
   790→ * Handles: accepted → preparing, preparing → ready_for_pickup.
   791→ *
   792→ * Security:
   793→ * - Validates order exists and current DB status
   794→ * - Validates transition is allowed
   795→ * - Race condition guard: UPDATE only if order_status still equals current
   796→ * - Records full audit trail in order_status_history
   797→ *
   798→ * @param orderId - UUID of the order
   799→ * @param targetStatus - The new status to transition to
   800→ * @param adminUserId - UUID of the admin making the change
   801→ */
   802→export async function updateOrderStatus(
   803→  orderId: string,
   804→  targetStatus: AdminOrderStatus,
   805→  adminUserId: string
   806→): Promise<UpdateOrderStatusResult> {
   807→  try {
   808→    const supabase = createAdminClient()
   809→
   810→    // 1. Validate inputs
   811→    if (!orderId || typeof orderId !== 'string') {
   812→      return { success: false, message: 'Invalid order ID' }
   813→    }
   814→    if (!targetStatus || typeof targetStatus !== 'string') {
   815→      return { success: false, message: 'Invalid target status' }
   816→    }
   817→
   818→    // 2. Validate target status is allowed in kitchen workflow
   819→    const allowedTargets = Object.values(KITCHEN_TRANSITIONS).flat()
   820→    if (!allowedTargets.includes(targetStatus)) {
   821→      return { success: false, message: 'Invalid status transition for this workflow' }
   822→    }
   823→
   824→    // 3. Fetch the current order
   825→    // eslint-disable-next-line @typescript-eslint/no-explicit-any
   826→    const { data: order, error: fetchErr }: { data: RawOrderRow | null; error: any } = await supabase
   827→      .from('orders')
   828→      .select('*')
   829→      .eq('id', orderId)
   830→      .single()
   831→
   832→    if (fetchErr || !order) {
   833→      return { success: false, message: 'Order not found' }
   834→    }
   835→
   836→    // 4. Determine what current status allows this transition
   837→    const currentStatus = order.order_status
   838→    const allowedFromCurrent = KITCHEN_TRANSITIONS[currentStatus]
   839→    if (!allowedFromCurrent || !allowedFromCurrent.includes(targetStatus)) {
   840→      // Provide a helpful message
   841→      const statusLabels: Record<string, string> = {
   842→        confirmed: 'Pending',
   843→        accepted: 'Accepted',
   844→        preparing: 'Preparing',
   845→        ready_for_pickup: 'Ready for Pickup',
   846→        completed: 'Completed',
   847→        cancelled: 'Cancelled',
   848→      }
   849→      const from = statusLabels[currentStatus] || currentStatus
   850→      const to = statusLabels[targetStatus] || targetStatus
   851→      return { success: false, message: `Cannot move order from "${from}" to "${to}"` }
   852→    }
   853→
   854→    // 5. Update with race condition guard (only update if status hasn't changed)
   855→    // eslint-disable-next-line @typescript-eslint/no-explicit-any
   856→    const sb = supabase as any
   857→    const { error: updateErr, count: updateCount } = await sb
   858→      .from('orders')
   859→      .update({
   860→        order_status: targetStatus,
   861→        updated_at: new Date().toISOString(),
   862→      })
   863→      .eq('id', orderId)
   864→      .eq('order_status', currentStatus)
   865→
   866→    if (updateErr) {
   867→      console.error('[ADMIN ORDER SERVICE] updateOrderStatus update error:', updateErr)
   868→      return { success: false, message: 'Failed to update order status. Please try again.' }
   869→    }
   870→
   871→    // 6. Check if the update actually happened (0 rows = race condition / already changed)
   872→    if (updateCount === 0) {
   873→      // Re-fetch to get current status for a helpful message
   874→      const { data: freshOrder } = await supabase
   875→        .from('orders')
   876→        .select('order_status')
   877→        .eq('id', orderId)
   878→        .single()
   879→
   880→      const statusLabels: Record<string, string> = {
   881→        accepted: 'Accepted', preparing: 'Preparing',
   882→        ready_for_pickup: 'Ready for Pickup', completed: 'Completed',
   883→        cancelled: 'Cancelled', confirmed: 'Pending',
   884→      }
   885→      const current = statusLabels[freshOrder?.order_status] || freshOrder?.order_status || 'Unknown'
   886→      return { success: false, message: `Order status was already updated to "${current}" by another action` }
   887→    }
   888→
   889→    // 7. Log status change in history
   890→    //    Table schema: (order_id, status, note, created_at)
   891→    //    Unique constraint: (order_id, status) — one row per status per order
   892→    //    Note encodes: previous → new | admin_id
   893→    const statusNote = `${currentStatus} → ${targetStatus} | admin: ${adminUserId}`
   894→    try {
   895→      const { error: histErr } = await sb.from('order_status_history').insert({
   896→        order_id: orderId,
   897→        status: targetStatus,
   898→        note: statusNote,
   899→      })
   900→      if (histErr && !String(histErr.message || histErr).includes('duplicate')) {
   901→        console.error('[ADMIN ORDER SERVICE] Failed to log status history:', histErr)
   902→      }
   903→    } catch (historyErr) {
   904→      console.error('[ADMIN ORDER SERVICE] Failed to log status history:', historyErr)
   905→    }
   906→
   907→    // 8. Success
   908→    const statusLabels: Record<string, string> = {
   909→      preparing: 'Preparing',
   910→      ready_for_pickup: 'Ready for Pickup',
   911→    }
   912→    const label = statusLabels[targetStatus] || targetStatus
   913→    return { success: true, message: `Order status updated to "${label}"` }
   914→  } catch (err) {
   915→    console.error('[ADMIN ORDER SERVICE] updateOrderStatus error:', err)
   916→    return { success: false, message: 'Something went wrong. Please try again.' }
   917→  }
   918→}

   922→// ============================================================================
   923→
   924→export function isValidTransition(
   925→  currentStatus: string,
   926→  targetStatus: string
   927→): boolean {
   928→  const allowed = VALID_TRANSITIONS[currentStatus]
   929→  if (!allowed) return false
   930→  return allowed.includes(targetStatus as AdminOrderStatus)
