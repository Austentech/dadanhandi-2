    51→/**
    52→ * Fetch dashboard statistics from the database.
    53→ * Returns counts for each order status category.
    54→ *
    55→ * IMPORTANT: "Pending" and "Upcoming" now use the preparation window.
    56→ *   - Pending (Current Queue): confirmed + paid + pickup slot within window
    57→ *   - Upcoming: confirmed + paid + today + pickup slot AFTER window
    58→ */
    59→export async function getDashboardStats(): Promise<DashboardStatsResult> {
    60→  try {
    61→    const supabase = createAdminClient()
    62→    const bounds = getPreparationWindowBounds()
    63→
    64→    // Get today's date range in IST
    65→    const now = new Date()
    66→    const istOffset = 5.5 * 60 * 60 * 1000
    67→    const istTime = new Date(now.getTime() + istOffset)
    68→    const todayStart = new Date(istTime.getFullYear(), istTime.getMonth(), istTime.getDate())
    69→    const todayStartUTC = new Date(todayStart.getTime() - istOffset)
    70→
    71→    const [todayRes, acceptedRes, preparingRes, readyRes, completedRes, cancelledRes] =
    72→      await Promise.all([
    73→        // Today's orders (all non-draft statuses)
    74→        supabase.from('orders').select('id', { count: 'exact', head: true })
    75→          .neq('order_status', 'draft')
    76→          .gte('created_at', todayStartUTC.toISOString()),
    77→        // Accepted
    78→        supabase.from('orders').select('id', { count: 'exact', head: true })
    79→          .eq('order_status', 'accepted'),
    80→        // Preparing
    81→        supabase.from('orders').select('id', { count: 'exact', head: true })
    82→          .eq('order_status', 'preparing'),
    83→        // Ready for pickup
    84→        supabase.from('orders').select('id', { count: 'exact', head: true })
    85→          .eq('order_status', 'ready_for_pickup'),
    86→        // Completed (today)
    87→        supabase.from('orders').select('id', { count: 'exact', head: true })
    88→          .eq('order_status', 'completed')
    89→          .gte('created_at', todayStartUTC.toISOString()),
    90→        // Cancelled (today)
    91→        supabase.from('orders').select('id', { count: 'exact', head: true })
    92→          .eq('order_status', 'cancelled')
    93→          .gte('created_at', todayStartUTC.toISOString()),
    94→      ])
    95→
    96→    // --- Pending (Current Queue): confirmed + paid + in preparation window + overdue ---
    97→    // Includes: orders in window + overdue (slot passed) + past-date unaccepted
    98→    // eslint-disable-next-line @typescript-eslint/no-explicit-any
    99→    const sb = supabase as any
   100→
   101→    const windowEndStr = `${String(Math.floor(bounds.windowEndMinutes / 60)).padStart(2, '0')}:${String(bounds.windowEndMinutes % 60).padStart(2, '0')}:00`
   102→
   103→    // Pending today: in-window + overdue (slot <= windowEnd, no lower bound)
   104→    const { count: pendingTodayCount } = await sb
   105→      .from('orders')
   106→      .select('id', { count: 'exact', head: true })
   107→      .eq('order_status', 'confirmed')
   108→      .eq('payment_status', 'succeeded')
   109→      .eq('pickup_date', bounds.todayDateStr)
   110→      .lte('pickup_slot_start', windowEndStr)
   111→
   112→    // Pending past: overdue from yesterday or earlier
   113→    const { count: pendingPastCount } = await sb
   114→      .from('orders')
   115→      .select('id', { count: 'exact', head: true })
   116→      .eq('order_status', 'confirmed')
   117→      .eq('payment_status', 'succeeded')
   118→      .lt('pickup_date', bounds.todayDateStr)
   119→
   120→    const pendingCount = (pendingTodayCount || 0) + (pendingPastCount || 0)
   121→
   122→    // --- Upcoming: confirmed + paid + today + AFTER preparation window ---
   123→    const { count: upcomingCount } = await sb
   124→      .from('orders')
   125→      .select('id', { count: 'exact', head: true })
   126→      .eq('order_status', 'confirmed')
   127→      .eq('payment_status', 'succeeded')
   128→      .eq('pickup_date', bounds.todayDateStr)
   129→      .gt('pickup_slot_start', windowEndStr)
   130→
   131→    return {
   132→      success: true,
   133→      stats: {
   134→        todayOrders: todayRes.count || 0,
   135→        pendingOrders: pendingCount || 0,
   136→        acceptedOrders: acceptedRes.count || 0,
   137→        preparingOrders: preparingRes.count || 0,
   138→        readyOrders: readyRes.count || 0,
   139→        completedOrders: completedRes.count || 0,
   140→        cancelledOrders: cancelledRes.count || 0,
   141→        upcomingOrders: upcomingCount || 0,
   142→      },
   143→    }
   144→  } catch (err) {
   145→    console.error('[ADMIN ORDER SERVICE] getDashboardStats error:', err)
   146→    return { success: false, stats: emptyStats() }
   147→  }
   148→}

   339→/**
   340→ * List orders with a given status, including items and customer info.
   341→ * For 'confirmed' status:
   342→ *   - Only shows PAID orders (payment_status='succeeded')
   343→ *   - Shows orders in the preparation window (current queue)
   344→ *   - Shows OVERDUE orders whose pickup slot has already passed (never accepted)
   345→ *   - Shows past-date orders (yesterday or earlier, still confirmed + paid)
   346→ *   - Orders with FUTURE pickup slots are HIDDEN until their window begins
   347→ *   - Each order includes an `isOverdue` flag computed server-side
   348→ */
   349→export async function listOrdersByStatus(
   350→  status: AdminOrderStatus,
   351→  options?: {
   352→    branchSlug?: string
   353→    search?: string
   354→    limit?: number
   355→    offset?: number
   356→  }
   357→): Promise<ListOrdersResult> {
   358→  try {
   359→    const supabase = createAdminClient()
   360→    const limit = options?.limit || 50
   361→    const offset = options?.offset || 0
   362→
   363→    // eslint-disable-next-line @typescript-eslint/no-explicit-any
   364→    const sb = supabase as any
   365→
   366→    const selectFields = `
   367→      *,
   368→      order_items (*)
   369→    `
   370→
   371→    let orders: RawOrderRow[]
   372→    let totalCount: number
   373→    // Store window bounds for isOverdue computation (only for confirmed)
   374→    let windowStartMinutes: number | null = null
   375→    let todayDateStr: string | null = null
   376→
   377→    if (status === 'confirmed') {
   378→      // --- Confirmed orders: two-query approach ---
   379→      // 1. Today's orders: in-window + overdue (no lower bound on slot)
   380→      // 2. Past-date overdue: pickup_date < today
   381→      // Future orders (slot after window end) are excluded.
   382→
   383→      const bounds = getPreparationWindowBounds()
   384→      windowStartMinutes = bounds.windowStartMinutes
   385→      todayDateStr = bounds.todayDateStr
   386→
   387→      const windowEndStr = `${String(Math.floor(bounds.windowEndMinutes / 60)).padStart(2, '0')}:${String(bounds.windowEndMinutes % 60).padStart(2, '0')}:00`
   388→
   389→      // Query 1: Today — in-window + overdue (slot <= windowEnd, no .gte lower bound)
   390→      let todayQ = sb
   391→        .from('orders')
   392→        .select(selectFields, { count: 'exact' })
   393→        .eq('order_status', 'confirmed')
   394→        .eq('payment_status', 'succeeded')
   395→        .eq('pickup_date', bounds.todayDateStr)
   396→        .lte('pickup_slot_start', windowEndStr)
   397→        .order('pickup_slot_start', { ascending: true })
   398→
   399→      // Query 2: Past-date overdue (yesterday or earlier, never accepted)
   400→      let pastQ = sb
   401→        .from('orders')
   402→        .select(selectFields)
   403→        .eq('order_status', 'confirmed')
   404→        .eq('payment_status', 'succeeded')
   405→        .lt('pickup_date', bounds.todayDateStr)
   406→        .order('pickup_date', { ascending: false })
   407→        .order('pickup_slot_start', { ascending: false })
   408→        .limit(50) // Safety limit for very old overdue orders
   409→
   410→      // Apply optional branch filter to both queries
   411→      if (options?.branchSlug) {
   412→        todayQ = todayQ.eq('branch_id', options.branchSlug)
   413→        pastQ = pastQ.eq('branch_id', options.branchSlug)
   414→      }
   415→
   416→      const [todayRes, pastRes] = await Promise.all([todayQ, pastQ])
   417→
   418→      if (todayRes.error) {
   419→        console.error('[ADMIN ORDER SERVICE] listOrdersByStatus today query error:', todayRes.error)
   420→        return { success: false, orders: [], totalCount: 0, error: 'Failed to fetch orders' }
   421→      }
   422→
   423→      const todayOrders = (todayRes.data || []) as RawOrderRow[]
   424→      const pastOrders = (pastRes.data || []) as RawOrderRow[]
   425→
   426→      // Merge: past overdue first (most recent past first), then today's
   427→      orders = [...pastOrders, ...todayOrders]
   428→      totalCount = (todayRes.count || 0) + pastOrders.length
   429→    } else {
   430→      // --- Non-confirmed: simple single query ---
   431→      let query = sb
   432→        .from('orders')
   433→        .select(selectFields, { count: 'exact' })
   434→        .eq('order_status', status)
   435→
   436→      query = query
   437→        .order('pickup_slot_start', { ascending: true })
   438→        .range(offset, offset + limit - 1)
   439→
   440→      if (options?.branchSlug) {
   441→        query = query.eq('branch_id', options.branchSlug)
   442→      }
   443→
   444→      const { data, error, count } = await query
   445→
   446→      if (error) {
   447→        console.error('[ADMIN ORDER SERVICE] listOrdersByStatus error:', error)
   448→        return { success: false, orders: [], totalCount: 0, error: 'Failed to fetch orders' }
   449→      }
   450→
   451→      orders = (data || []) as RawOrderRow[]
   452→      totalCount = count || 0
   453→    }
   454→
   455→    // Apply pagination (for confirmed, we already have all orders merged)
   456→    if (status === 'confirmed') {
   457→      orders = orders.slice(offset, offset + limit)
   458→    }
   459→
   460→    if (!orders || orders.length === 0) {
   461→      return { success: true, orders: [], totalCount: 0 }
   462→    }
   463→
   464→    // Fetch customer profiles for all orders
   465→    // CRITICAL: profiles.id ≠ auth.users.id. The FK is profiles.auth_user_id → auth.users(id).
   466→    // orders.user_id references auth.users(id), so we must match on auth_user_id.
   467→    const userIds = [...new Set(orders.map((o: RawOrderRow) => o.user_id))]
   468→    const { data: profiles } = await supabase
   469→      .from('profiles')
   470→      .select('id, auth_user_id, full_name, whatsapp_number, mobile_number')
   471→      .in('auth_user_id', userIds)
   472→
   473→    const profileMap = new Map<string, { name: string; whatsappNumber: string | null; mobileNumber: string | null }>()
   474→    for (const p of (profiles || []) as RawProfileRow[]) {
   475→      profileMap.set(p.auth_user_id, {
   476→        name: p.full_name || 'Customer',
   477→        whatsappNumber: p.whatsapp_number || null,
   478→        mobileNumber: p.mobile_number || null,
   479→      })
   480→    }
   481→
   482→    // Fetch branch info separately (avoid unreliable PostgREST join key naming)
   483→    const branchIds = [...new Set(orders.map((o: RawOrderRow) => o.branch_id))]
   484→    const { data: branches } = await supabase
   485→      .from('branches')
   486→      .select('id, name, slug, city')
   487→      .in('id', branchIds)
   488→
   489→    const branchMap = new Map<string, { name: string; slug: string; city: string }>()
   490→    for (const b of (branches || []) as { id: string; name: string; slug: string; city: string }[]) {
   491→      branchMap.set(b.id, { name: b.name, slug: b.slug, city: b.city })
   492→    }
   493→
   494→    // Search filter (applied in-memory since Supabase free tier doesn't support full-text across joins)
   495→    let filteredOrders = orders as RawOrderRow[]
   496→    if (options?.search) {
   497→      const q = options.search.toLowerCase()
   498→      filteredOrders = filteredOrders.filter((o) => {
   499→        const orderNum = (o.order_number || '').toLowerCase()
   500→        const profile = profileMap.get(o.user_id)
   501→        const customerName = (profile?.name || '').toLowerCase()
   502→        return orderNum.includes(q) || customerName.includes(q)
   503→      })
   504→    }
   505→
   506→    // Helper: compute isOverdue for a confirmed order (server-side, using window bounds)
   507→    const computeIsOverdue = (raw: RawOrderRow): boolean => {
   508→      if (windowStartMinutes === null || todayDateStr === null) return false
   509→      // Past-date orders are always overdue
   510→      if (raw.pickup_date < todayDateStr) return true
   511→      // Today's orders: overdue if slot started before the window
   512→      const [h, m] = raw.pickup_slot_start.split(':').map(Number)
   513→      if (isNaN(h) || isNaN(m)) return false
   514→      const slotMinutes = h * 60 + m
   515→      return slotMinutes < windowStartMinutes
   516→    }
   517→
   518→    // Map to typed response with proper snake_case -> camelCase conversion
   519→    const mapped: AdminOrderWithItems[] = filteredOrders.map((o) => {
   520→      const profile = profileMap.get(o.user_id)
   521→      const branch = branchMap.get(o.branch_id)
   522→      return {
   523→        ...mapOrderRow(o),
   524→        items: (o.order_items || []).map(mapOrderItemRow),
   525→        branch: branch || null,
   526→        customer: profile || { name: 'Customer', whatsappNumber: null, mobileNumber: null },
   527→        isOverdue: status === 'confirmed' ? computeIsOverdue(o) : undefined,
   528→      }
   529→    })
   530→
   531→    return {
   532→      success: true,
   533→      orders: mapped,
   534→      totalCount,
   535→    }
   536→  } catch (err) {
   537→    console.error('[ADMIN ORDER SERVICE] listOrdersByStatus error:', err)
   538→    return { success: false, orders: [], totalCount: 0, error: 'Failed to fetch orders' }
   539→  }
   540→}

   553→/**
   554→ * Accept an order: transition from 'confirmed' to 'accepted'.
   555→ * - Validates order exists, is 'confirmed', and payment succeeded
   556→ * - Prevents duplicate acceptance (idempotent by design)
   557→ * - Records status change in order_status_history
   558→ */
   559→export async function acceptOrder(
   560→  orderId: string,
   561→  adminUserId: string
   562→): Promise<AcceptOrderResult> {
   563→  try {
   564→    const supabase = createAdminClient()
   565→
   566→    // 1. Validate input
   567→    if (!orderId || typeof orderId !== 'string') {
   568→      return { success: false, message: 'Invalid order ID' }
   569→    }
   570→
   571→    // 2. Fetch the order
   572→    // eslint-disable-next-line @typescript-eslint/no-explicit-any
   573→    const { data: order, error: fetchErr }: { data: RawOrderRow | null; error: any } = await supabase
   574→      .from('orders')
   575→      .select('*')
   576→      .eq('id', orderId)
   577→      .single()
   578→
   579→    if (fetchErr || !order) {
   580→      return { success: false, message: 'Order not found' }
   581→    }
   582→
   583→    // 3. Validate payment status — must be succeeded
   584→    if (order.payment_status !== 'succeeded') {
   585→      return { success: false, message: 'Cannot accept an unpaid order' }
   586→    }
   587→
   588→    // 4. Validate current status
   589→    if (order.order_status !== 'confirmed') {
   590→      if (order.order_status === 'accepted') {
   591→        return { success: true, message: 'Order already accepted' }
   592→      }
   593→      return { success: false, message: `Cannot accept order in '${order.order_status}' status` }
   594→    }
   595→
   596→    // 5. Validate the transition is allowed
   597→    const allowedTransitions = VALID_TRANSITIONS[order.order_status] || []
   598→    if (!allowedTransitions.includes('accepted')) {
   599→      return { success: false, message: 'Invalid status transition' }
   600→    }
   601→
   602→    // 6. Update order status (race condition guard: only update if still confirmed)
   603→    // eslint-disable-next-line @typescript-eslint/no-explicit-any
   604→    const sb = supabase as any
   605→    const { error: updateErr } = await sb
   606→      .from('orders')
   607→      .update({
   608→        order_status: 'accepted',
   609→        updated_at: new Date().toISOString(),
   610→      })
   611→      .eq('id', orderId)
   612→      .eq('order_status', 'confirmed')
   613→
   614→    if (updateErr) {
   615→      console.error('[ADMIN ORDER SERVICE] acceptOrder update error:', updateErr)
   616→      return { success: false, message: 'Failed to update order. Please try again.' }
   617→    }
   618→
   619→    // 7. Log status change in history
   620→    //    Table schema: (order_id, status, note, created_at)
   621→    //    Unique constraint: (order_id, status) — one row per status per order
   622→    try {
   623→      const { error: histErr } = await sb.from('order_status_history').insert({
   624→        order_id: orderId,
   625→        status: 'accepted',
   626→        note: 'Accepted by admin',
   627→      })
   628→      // Ignore duplicate key errors (order already has 'accepted' status row)
   629→      if (histErr && !String(histErr.message || histErr).includes('duplicate')) {
   630→        console.error('[ADMIN ORDER SERVICE] Failed to log status history:', histErr)
   631→      }
   632→    } catch (historyErr) {
   633→      // Log but don't fail — the order status was already updated
   634→      console.error('[ADMIN ORDER SERVICE] Failed to log status history:', historyErr)
   635→    }
   636→
   637→    return { success: true, message: 'Order accepted successfully' }
   638→  } catch (err) {
   639→    console.error('[ADMIN ORDER SERVICE] acceptOrder error:', err)
   640→    return { success: false, message: 'Something went wrong. Please try again.' }
   641→  }
   642→}

   648→/**
   649→ * List all ongoing orders (accepted, preparing, ready_for_pickup).
   650→ * Includes items, customer info, branch info, and accepted-at timestamp.
   651→ * Ordered by pickup time (soonest first), then by status priority.
   652→ */
   653→export async function listOngoingOrders(options?: {
   654→  branchSlug?: string
   655→  search?: string
   656→  limit?: number
   657→  offset?: number
   658→}): Promise<ListOrdersResult> {
   659→  try {
   660→    const supabase = createAdminClient()
   661→    const limit = options?.limit || 50
   662→    const offset = options?.offset || 0
   663→
   664→    // eslint-disable-next-line @typescript-eslint/no-explicit-any
   665→    const sb = supabase as any
   666→
   667→    const selectFields = `
   668→      *,
   669→      order_items (*)
   670→    `
   671→
   672→    let query = sb
   673→      .from('orders')
   674→      .select(selectFields, { count: 'exact' })
   675→      .in('order_status', ['accepted', 'preparing', 'ready_for_pickup'])
   676→      .order('pickup_slot_start', { ascending: true })
   677→      .range(offset, offset + limit - 1)
   678→
   679→    if (options?.branchSlug) {
   680→      query = query.eq('branch_id', options.branchSlug)
   681→    }
   682→
   683→    const { data: orders, error, count } = await query
   684→
   685→    if (error) {
   686→      console.error('[ADMIN ORDER SERVICE] listOngoingOrders error:', error)
   687→      return { success: false, orders: [], totalCount: 0, error: 'Failed to fetch ongoing orders' }
   688→    }
   689→
   690→    if (!orders || orders.length === 0) {
   691→      return { success: true, orders: [], totalCount: 0 }
   692→    }
   693→
   694→    const rawOrders = orders as RawOrderRow[]
   695→
   696→    // --- Batch fetch customer profiles ---
   697→    // CRITICAL: profiles.id ≠ auth.users.id. Must join on auth_user_id.
   698→    const userIds = [...new Set(rawOrders.map((o) => o.user_id))]
   699→    const { data: profiles } = await supabase
   700→      .from('profiles')
   701→      .select('id, auth_user_id, full_name, whatsapp_number, mobile_number')
   702→      .in('auth_user_id', userIds)
   703→
   704→    const profileMap = new Map<string, { name: string; whatsappNumber: string | null; mobileNumber: string | null }>()
   705→    for (const p of (profiles || []) as RawProfileRow[]) {
   706→      profileMap.set(p.auth_user_id, {
   707→        name: p.full_name || 'Customer',
   708→        whatsappNumber: p.whatsapp_number || null,
   709→        mobileNumber: p.mobile_number || null,
   710→      })
   711→    }
   712→
   713→    // --- Batch fetch branch info (separate query, not PostgREST join) ---
   714→    const branchIdSet = [...new Set(rawOrders.map((o) => o.branch_id))]
   715→    const { data: branchRows } = await supabase
   716→      .from('branches')
   717→      .select('id, name, slug, city')
   718→      .in('id', branchIdSet)
   719→
   720→    const branchMap = new Map<string, { name: string; slug: string; city: string }>()
   721→    for (const b of (branchRows || []) as { id: string; name: string; slug: string; city: string }[]) {
   722→      branchMap.set(b.id, { name: b.name, slug: b.slug, city: b.city })
   723→    }
   724→
   725→    // --- Batch fetch accepted-at timestamps from status history ---
   726→    const orderIds = rawOrders.map((o) => o.id)
   727→    const { data: historyRows } = await supabase
   728→      .from('order_status_history')
   729→      .select('order_id, created_at')
   730→      .in('order_id', orderIds)
   731→      .eq('status', 'accepted')
   732→
   733→    const acceptedAtMap = new Map<string, string>()
   734→    for (const row of (historyRows || []) as { order_id: string; created_at: string }[]) {
   735→      if (!acceptedAtMap.has(row.order_id)) {
   736→        acceptedAtMap.set(row.order_id, row.created_at)
   737→      }
   738→    }
   739→
   740→    // --- Search filter (in-memory) ---
   741→    let filteredOrders = rawOrders
   742→    if (options?.search) {
   743→      const q = options.search.toLowerCase()
   744→      filteredOrders = filteredOrders.filter((o) => {
   745→        const orderNum = (o.order_number || '').toLowerCase()
   746→        const profile = profileMap.get(o.user_id)
   747→        const customerName = (profile?.name || '').toLowerCase()
   748→        return orderNum.includes(q) || customerName.includes(q)
   749→      })
   750→    }
   751→
   752→    // --- Status priority for sorting (accepted=1, preparing=2, ready=3) ---
   753→    const statusWeight: Record<string, number> = {
   754→      accepted: 1,
   755→      preparing: 2,
   756→      ready_for_pickup: 3,
   757→    }
   758→
   759→    // --- Map to typed response ---
   760→    const mapped: AdminOrderWithItems[] = filteredOrders.map((o) => {
   761→      const profile = profileMap.get(o.user_id)
   762→      const branch = branchMap.get(o.branch_id)
   763→      return {
   764→        ...mapOrderRow(o),
   765→        items: (o.order_items || []).map(mapOrderItemRow),
   766→        branch: branch || null,
   767→        customer: profile || { name: 'Customer', whatsappNumber: null, mobileNumber: null },
   768→        acceptedAt: acceptedAtMap.get(o.id) || undefined,
   769→      }
   770→    })
   771→
   772→    // Sort: pickup time ascending, then status weight ascending (accepted before preparing)
   773→    mapped.sort((a, b) => {
   774→      const slotA = a.pickupSlotStart
   775→      const slotB = b.pickupSlotStart
   776→      if (slotA !== slotB) return slotA.localeCompare(slotB)
   777→      return (statusWeight[a.orderStatus] || 99) - (statusWeight[b.orderStatus] || 99)
   778→    })
   779→
   780→    return {
   781→      success: true,
   782→      orders: mapped,
   783→      totalCount: count || 0,
   784→    }
   785→  } catch (err) {
   786→    console.error('[ADMIN ORDER SERVICE] listOngoingOrders error:', err)
   787→    return { success: false, orders: [], totalCount: 0, error: 'Failed to fetch ongoing orders' }
   788→  }
   789→}

   791→// ============================================================================
   792→// UPDATE ORDER STATUS (Kitchen Workflow)
   793→// ============================================================================
   794→
   795→export interface UpdateOrderStatusResult {
   796→  success: boolean
   797→  message: string
   798→  error?: string
   799→}
   800→
   801→/**
   802→ * Subset of VALID_TRANSITIONS for the kitchen workflow.
   803→ * This module only handles forward transitions:
   804→ *   accepted → preparing → ready_for_pickup
   805→ * Cancellation and other transitions are handled by other modules.
   806→ */
   807→const KITCHEN_TRANSITIONS: Record<string, AdminOrderStatus[]> = {
   808→  accepted: ['preparing'],
   809→  preparing: ['ready_for_pickup'],
   810→}
   811→
   812→/**
   813→ * Update an order's status as part of the kitchen workflow.
   814→ * Handles: accepted → preparing, preparing → ready_for_pickup.
   815→ *
   816→ * Security:
   817→ * - Validates order exists and current DB status
   818→ * - Validates transition is allowed
   819→ * - Race condition guard: UPDATE only if order_status still equals current
   820→ * - Records full audit trail in order_status_history
   821→ *
   822→ * @param orderId - UUID of the order
   823→ * @param targetStatus - The new status to transition to
   824→ * @param adminUserId - UUID of the admin making the change
   825→ */
   826→export async function updateOrderStatus(
   827→  orderId: string,
   828→  targetStatus: AdminOrderStatus,
   829→  adminUserId: string
   830→): Promise<UpdateOrderStatusResult> {
   831→  try {
   832→    const supabase = createAdminClient()
   833→
   834→    // 1. Validate inputs
   835→    if (!orderId || typeof orderId !== 'string') {
   836→      return { success: false, message: 'Invalid order ID' }
   837→    }
   838→    if (!targetStatus || typeof targetStatus !== 'string') {
   839→      return { success: false, message: 'Invalid target status' }
   840→    }
   841→
   842→    // 2. Validate target status is allowed in kitchen workflow
   843→    const allowedTargets = Object.values(KITCHEN_TRANSITIONS).flat()
   844→    if (!allowedTargets.includes(targetStatus)) {
   845→      return { success: false, message: 'Invalid status transition for this workflow' }
   846→    }
   847→
   848→    // 3. Fetch the current order
   849→    // eslint-disable-next-line @typescript-eslint/no-explicit-any
   850→    const { data: order, error: fetchErr }: { data: RawOrderRow | null; error: any } = await supabase
   851→      .from('orders')
   852→      .select('*')
   853→      .eq('id', orderId)
   854→      .single()
   855→
   856→    if (fetchErr || !order) {
   857→      return { success: false, message: 'Order not found' }
   858→    }
   859→
   860→    // 4. Determine what current status allows this transition
   861→    const currentStatus = order.order_status
   862→    const allowedFromCurrent = KITCHEN_TRANSITIONS[currentStatus]
   863→    if (!allowedFromCurrent || !allowedFromCurrent.includes(targetStatus)) {
   864→      // Provide a helpful message
   865→      const statusLabels: Record<string, string> = {
   866→        confirmed: 'Pending',
   867→        accepted: 'Accepted',
   868→        preparing: 'Preparing',
   869→        ready_for_pickup: 'Ready for Pickup',
   870→        completed: 'Completed',
   871→        cancelled: 'Cancelled',
   872→      }
   873→      const from = statusLabels[currentStatus] || currentStatus
   874→      const to = statusLabels[targetStatus] || targetStatus
   875→      return { success: false, message: `Cannot move order from "${from}" to "${to}"` }
   876→    }
   877→
   878→    // 5. Update with race condition guard (only update if status hasn't changed)
   879→    // eslint-disable-next-line @typescript-eslint/no-explicit-any
   880→    const sb = supabase as any
   881→    const { error: updateErr, count: updateCount } = await sb
   882→      .from('orders')
   883→      .update({
   884→        order_status: targetStatus,
   885→        updated_at: new Date().toISOString(),
   886→      })
   887→      .eq('id', orderId)
   888→      .eq('order_status', currentStatus)
   889→
   890→    if (updateErr) {
   891→      console.error('[ADMIN ORDER SERVICE] updateOrderStatus update error:', updateErr)
   892→      return { success: false, message: 'Failed to update order status. Please try again.' }
   893→    }
   894→
   895→    // 6. Check if the update actually happened (0 rows = race condition / already changed)
   896→    if (updateCount === 0) {
   897→      // Re-fetch to get current status for a helpful message
   898→      const { data: freshOrder } = await supabase
   899→        .from('orders')
   900→        .select('order_status')
   901→        .eq('id', orderId)
   902→        .single()
   903→
   904→      const statusLabels: Record<string, string> = {
   905→        accepted: 'Accepted', preparing: 'Preparing',
   906→        ready_for_pickup: 'Ready for Pickup', completed: 'Completed',
   907→        cancelled: 'Cancelled', confirmed: 'Pending',
   908→      }
   909→      const current = statusLabels[freshOrder?.order_status] || freshOrder?.order_status || 'Unknown'
   910→      return { success: false, message: `Order status was already updated to "${current}" by another action` }
   911→    }
   912→
   913→    // 7. Log status change in history
   914→    //    Table schema: (order_id, status, note, created_at)
   915→    //    Unique constraint: (order_id, status) — one row per status per order
   916→    //    Note encodes: previous → new | admin_id
   917→    const statusNote = `${currentStatus} → ${targetStatus} | admin: ${adminUserId}`
   918→    try {
   919→      const { error: histErr } = await sb.from('order_status_history').insert({
   920→        order_id: orderId,
   921→        status: targetStatus,
   922→        note: statusNote,
   923→      })
   924→      if (histErr && !String(histErr.message || histErr).includes('duplicate')) {
   925→        console.error('[ADMIN ORDER SERVICE] Failed to log status history:', histErr)
   926→      }
   927→    } catch (historyErr) {
   928→      console.error('[ADMIN ORDER SERVICE] Failed to log status history:', historyErr)
   929→    }
   930→
   931→    // 8. Success
   932→    const statusLabels: Record<string, string> = {
   933→      preparing: 'Preparing',
   934→      ready_for_pickup: 'Ready for Pickup',
   935→    }
   936→    const label = statusLabels[targetStatus] || targetStatus
   937→    return { success: true, message: `Order status updated to "${label}"` }
   938→  } catch (err) {
   939→    console.error('[ADMIN ORDER SERVICE] updateOrderStatus error:', err)
   940→    return { success: false, message: 'Something went wrong. Please try again.' }
   941→  }
   942→}

   945→// VALIDATE STATUS TRANSITION (reusable)
   946→// ============================================================================
   947→
   948→export function isValidTransition(
   949→  currentStatus: string,
   950→  targetStatus: string
   951→): boolean {
   952→  const allowed = VALID_TRANSITIONS[currentStatus]
   953→  if (!allowed) return false
   954→  return allowed.includes(targetStatus as AdminOrderStatus)
