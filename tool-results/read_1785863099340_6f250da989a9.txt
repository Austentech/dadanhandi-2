    33→    33→  accepted: ['preparing', 'cancelled'],

    51→    51→/**
    52→    52→ * Fetch dashboard statistics from the database.
    53→    53→ * Returns counts for each order status category.
    54→    54→ *
    55→    55→ * IMPORTANT: "Pending" and "Upcoming" now use the preparation window.
    56→    56→ *   - Pending (Current Queue): confirmed + paid + pickup slot within window
    57→    57→ *   - Upcoming: confirmed + paid + today + pickup slot AFTER window
    58→    58→ */
    59→    59→export async function getDashboardStats(): Promise<DashboardStatsResult> {
    60→    60→  try {
    61→    61→    const supabase = createAdminClient()
    62→    62→    const bounds = getPreparationWindowBounds()
    63→    63→
    64→    64→    // Get today's date range in IST
    65→    65→    const now = new Date()
    66→    66→    const istOffset = 5.5 * 60 * 60 * 1000
    67→    67→    const istTime = new Date(now.getTime() + istOffset)
    68→    68→    const todayStart = new Date(istTime.getFullYear(), istTime.getMonth(), istTime.getDate())
    69→    69→    const todayStartUTC = new Date(todayStart.getTime() - istOffset)
    70→    70→
    71→    71→    const [todayRes, acceptedRes, preparingRes, readyRes, completedRes, cancelledRes] =
    72→    72→      await Promise.all([
    73→    73→        // Today's orders (all non-draft statuses)
    74→    74→        supabase.from('orders').select('id', { count: 'exact', head: true })
    75→    75→          .neq('order_status', 'draft')
    76→    76→          .gte('created_at', todayStartUTC.toISOString()),
    77→    77→        // Accepted
    78→    78→        supabase.from('orders').select('id', { count: 'exact', head: true })
    79→    79→          .eq('order_status', 'accepted'),
    80→    80→        // Preparing
    81→    81→        supabase.from('orders').select('id', { count: 'exact', head: true })
    82→    82→          .eq('order_status', 'preparing'),
    83→    83→        // Ready for pickup
    84→    84→        supabase.from('orders').select('id', { count: 'exact', head: true })
    85→    85→          .eq('order_status', 'ready_for_pickup'),
    86→    86→        // Completed (today)
    87→    87→        supabase.from('orders').select('id', { count: 'exact', head: true })
    88→    88→          .eq('order_status', 'completed')
    89→    89→          .gte('created_at', todayStartUTC.toISOString()),
    90→    90→        // Cancelled (today)
    91→    91→        supabase.from('orders').select('id', { count: 'exact', head: true })
    92→    92→          .eq('order_status', 'cancelled')
    93→    93→          .gte('created_at', todayStartUTC.toISOString()),
    94→    94→      ])
    95→    95→
    96→    96→    // --- Pending (Current Queue): confirmed + paid + in preparation window + overdue ---
    97→    97→    // Includes: orders in window + overdue (slot passed) + past-date unaccepted
    98→    98→    // eslint-disable-next-line @typescript-eslint/no-explicit-any
    99→    99→    const sb = supabase as any
   100→   100→
   101→   101→    const windowEndStr = `${String(Math.floor(bounds.windowEndMinutes / 60)).padStart(2, '0')}:${String(bounds.windowEndMinutes % 60).padStart(2, '0')}:00`
   102→   102→
   103→   103→    // Pending today: in-window + overdue (slot <= windowEnd, no lower bound)
   104→   104→    const { count: pendingTodayCount } = await sb
   105→   105→      .from('orders')
   106→   106→      .select('id', { count: 'exact', head: true })
   107→   107→      .eq('order_status', 'confirmed')
   108→   108→      .eq('payment_status', 'succeeded')
   109→   109→      .eq('pickup_date', bounds.todayDateStr)
   110→   110→      .lte('pickup_slot_start', windowEndStr)
   111→   111→
   112→   112→    // Pending past: overdue from yesterday or earlier
   113→   113→    const { count: pendingPastCount } = await sb
   114→   114→      .from('orders')
   115→   115→      .select('id', { count: 'exact', head: true })
   116→   116→      .eq('order_status', 'confirmed')
   117→   117→      .eq('payment_status', 'succeeded')
   118→   118→      .lt('pickup_date', bounds.todayDateStr)
   119→   119→
   120→   120→    const pendingCount = (pendingTodayCount || 0) + (pendingPastCount || 0)
   121→   121→
   122→   122→    // --- Upcoming: confirmed + paid + today + AFTER preparation window ---
   123→   123→    const { count: upcomingCount } = await sb
   124→   124→      .from('orders')
   125→   125→      .select('id', { count: 'exact', head: true })
   126→   126→      .eq('order_status', 'confirmed')
   127→   127→      .eq('payment_status', 'succeeded')
   128→   128→      .eq('pickup_date', bounds.todayDateStr)
   129→   129→      .gt('pickup_slot_start', windowEndStr)
   130→   130→
   131→   131→    return {
   132→   132→      success: true,
   133→   133→      stats: {
   134→   134→        todayOrders: todayRes.count || 0,
   135→   135→        pendingOrders: pendingCount || 0,
   136→   136→        acceptedOrders: acceptedRes.count || 0,
   137→   137→        preparingOrders: preparingRes.count || 0,
   138→   138→        readyOrders: readyRes.count || 0,
   139→   139→        completedOrders: completedRes.count || 0,
   140→   140→        cancelledOrders: cancelledRes.count || 0,
   141→   141→        upcomingOrders: upcomingCount || 0,
   142→   142→      },
   143→   143→    }
   144→   144→  } catch (err) {
   145→   145→    console.error('[ADMIN ORDER SERVICE] getDashboardStats error:', err)
   146→   146→    return { success: false, stats: emptyStats() }
   147→   147→  }
   148→   148→}

   350→   350→/**
   351→   351→ * List orders with a given status, including items and customer info.
   352→   352→ * For 'confirmed' status:
   353→   353→ *   - Only shows PAID orders (payment_status='succeeded')
   354→   354→ *   - Shows orders in the preparation window (current queue)
   355→   355→ *   - Shows OVERDUE orders whose pickup slot has already passed (never accepted)
   356→   356→ *   - Shows past-date orders (yesterday or earlier, still confirmed + paid)
   357→   357→ *   - Orders with FUTURE pickup slots are HIDDEN until their window begins
   358→   358→ *   - Each order includes an `isOverdue` flag computed server-side
   359→   359→ */
   360→   360→export async function listOrdersByStatus(
   361→   361→  status: AdminOrderStatus,
   362→   362→  options?: {
   363→   363→    branchSlug?: string
   364→   364→    search?: string
   365→   365→    limit?: number
   366→   366→    offset?: number
   367→   367→  }
   368→   368→): Promise<ListOrdersResult> {
   369→   369→  try {
   370→   370→    const supabase = createAdminClient()
   371→   371→    const limit = options?.limit || 50
   372→   372→    const offset = options?.offset || 0
   373→   373→
   374→   374→    // eslint-disable-next-line @typescript-eslint/no-explicit-any
   375→   375→    const sb = supabase as any
   376→   376→
   377→   377→    const selectFields = `
   378→   378→      *,
   379→   379→      order_items (*)
   380→   380→    `
   381→   381→
   382→   382→    let orders: RawOrderRow[]
   383→   383→    let totalCount: number
   384→   384→    // Store window bounds for isOverdue computation (only for confirmed)
   385→   385→    let windowStartMinutes: number | null = null
   386→   386→    let todayDateStr: string | null = null
   387→   387→
   388→   388→    if (status === 'confirmed') {
   389→   389→      // --- Confirmed orders: two-query approach ---
   390→   390→      // 1. Today's orders: in-window + overdue (no lower bound on slot)
   391→   391→      // 2. Past-date overdue: pickup_date < today
   392→   392→      // Future orders (slot after window end) are excluded.
   393→   393→
   394→   394→      const bounds = getPreparationWindowBounds()
   395→   395→      windowStartMinutes = bounds.windowStartMinutes
   396→   396→      todayDateStr = bounds.todayDateStr
   397→   397→
   398→   398→      const windowEndStr = `${String(Math.floor(bounds.windowEndMinutes / 60)).padStart(2, '0')}:${String(bounds.windowEndMinutes % 60).padStart(2, '0')}:00`
   399→   399→
   400→   400→      // Query 1: Today — in-window + overdue (slot <= windowEnd, no .gte lower bound)
   401→   401→      let todayQ = sb
   402→   402→        .from('orders')
   403→   403→        .select(selectFields, { count: 'exact' })
   404→   404→        .eq('order_status', 'confirmed')
   405→   405→        .eq('payment_status', 'succeeded')
   406→   406→        .eq('pickup_date', bounds.todayDateStr)
   407→   407→        .lte('pickup_slot_start', windowEndStr)
   408→   408→        .order('pickup_slot_start', { ascending: true })
   409→   409→
   410→   410→      // Query 2: Past-date overdue (yesterday or earlier, never accepted)
   411→   411→      let pastQ = sb
   412→   412→        .from('orders')
   413→   413→        .select(selectFields)
   414→   414→        .eq('order_status', 'confirmed')
   415→   415→        .eq('payment_status', 'succeeded')
   416→   416→        .lt('pickup_date', bounds.todayDateStr)
   417→   417→        .order('pickup_date', { ascending: false })
   418→   418→        .order('pickup_slot_start', { ascending: false })
   419→   419→        .limit(50) // Safety limit for very old overdue orders
   420→   420→
   421→   421→      // Apply optional branch filter to both queries
   422→   422→      if (options?.branchSlug) {
   423→   423→        todayQ = todayQ.eq('branch_id', options.branchSlug)
   424→   424→        pastQ = pastQ.eq('branch_id', options.branchSlug)
   425→   425→      }
   426→   426→
   427→   427→      const [todayRes, pastRes] = await Promise.all([todayQ, pastQ])
   428→   428→
   429→   429→      if (todayRes.error) {
   430→   430→        console.error('[ADMIN ORDER SERVICE] listOrdersByStatus today query error:', todayRes.error)
   431→   431→        return { success: false, orders: [], totalCount: 0, error: 'Failed to fetch orders' }
   432→   432→      }
   433→   433→
   434→   434→      const todayOrders = (todayRes.data || []) as RawOrderRow[]
   435→   435→      const pastOrders = (pastRes.data || []) as RawOrderRow[]
   436→   436→
   437→   437→      // Merge: past overdue first (most recent past first), then today's
   438→   438→      orders = [...pastOrders, ...todayOrders]
   439→   439→      totalCount = (todayRes.count || 0) + pastOrders.length
   440→   440→    } else {
   441→   441→      // --- Non-confirmed: simple single query ---
   442→   442→      let query = sb
   443→   443→        .from('orders')
   444→   444→        .select(selectFields, { count: 'exact' })
   445→   445→        .eq('order_status', status)
   446→   446→
   447→   447→      query = query
   448→   448→        .order('pickup_slot_start', { ascending: true })
   449→   449→        .range(offset, offset + limit - 1)
   450→   450→
   451→   451→      if (options?.branchSlug) {
   452→   452→        query = query.eq('branch_id', options.branchSlug)
   453→   453→      }
   454→   454→
   455→   455→      const { data, error, count } = await query
   456→   456→
   457→   457→      if (error) {
   458→   458→        console.error('[ADMIN ORDER SERVICE] listOrdersByStatus error:', error)
   459→   459→        return { success: false, orders: [], totalCount: 0, error: 'Failed to fetch orders' }
   460→   460→      }
   461→   461→
   462→   462→      orders = (data || []) as RawOrderRow[]
   463→   463→      totalCount = count || 0
   464→   464→    }
   465→   465→
   466→   466→    // Apply pagination (for confirmed, we already have all orders merged)
   467→   467→    if (status === 'confirmed') {
   468→   468→      orders = orders.slice(offset, offset + limit)
   469→   469→    }
   470→   470→
   471→   471→    if (!orders || orders.length === 0) {
   472→   472→      return { success: true, orders: [], totalCount: 0 }
   473→   473→    }
   474→   474→
   475→   475→    // Fetch customer profiles for all orders
   476→   476→    // CRITICAL: profiles.id ≠ auth.users.id. The FK is profiles.auth_user_id → auth.users(id).
   477→   477→    // orders.user_id references auth.users(id), so we must match on auth_user_id.
   478→   478→    const userIds = [...new Set(orders.map((o: RawOrderRow) => o.user_id))]
   479→   479→    const { data: profiles } = await supabase
   480→   480→      .from('profiles')
   481→   481→      .select('id, auth_user_id, full_name, whatsapp_number, mobile_number')
   482→   482→      .in('auth_user_id', userIds)
   483→   483→
   484→   484→    const profileMap = new Map<string, { name: string; whatsappNumber: string | null; mobileNumber: string | null }>()
   485→   485→    for (const p of (profiles || []) as RawProfileRow[]) {
   486→   486→      profileMap.set(p.auth_user_id, {
   487→   487→        name: p.full_name || 'Customer',
   488→   488→        whatsappNumber: p.whatsapp_number || null,
   489→   489→        mobileNumber: p.mobile_number || null,
   490→   490→      })
   491→   491→    }
   492→   492→
   493→   493→    // Fetch branch info separately (avoid unreliable PostgREST join key naming)
   494→   494→    const branchIds = [...new Set(orders.map((o: RawOrderRow) => o.branch_id))]
   495→   495→    const { data: branches } = await supabase
   496→   496→      .from('branches')
   497→   497→      .select('id, name, slug, city')
   498→   498→      .in('id', branchIds)
   499→   499→
   500→   500→    const branchMap = new Map<string, { name: string; slug: string; city: string }>()
   501→   501→    for (const b of (branches || []) as { id: string; name: string; slug: string; city: string }[]) {
   502→   502→      branchMap.set(b.id, { name: b.name, slug: b.slug, city: b.city })
   503→   503→    }
   504→   504→
   505→   505→    // Search filter (applied in-memory since Supabase free tier doesn't support full-text across joins)
   506→   506→    let filteredOrders = orders as RawOrderRow[]
   507→   507→    if (options?.search) {
   508→   508→      const q = options.search.toLowerCase()
   509→   509→      filteredOrders = filteredOrders.filter((o) => {
   510→   510→        const orderNum = (o.order_number || '').toLowerCase()
   511→   511→        const profile = profileMap.get(o.user_id)
   512→   512→        const customerName = (profile?.name || '').toLowerCase()
   513→   513→        return orderNum.includes(q) || customerName.includes(q)
   514→   514→      })
   515→   515→    }
   516→   516→
   517→   517→    // Helper: compute isOverdue for a confirmed order (server-side, using window bounds)
   518→   518→    const computeIsOverdue = (raw: RawOrderRow): boolean => {
   519→   519→      if (windowStartMinutes === null || todayDateStr === null) return false
   520→   520→      // Past-date orders are always overdue
   521→   521→      if (raw.pickup_date < todayDateStr) return true
   522→   522→      // Today's orders: overdue if slot started before the window
   523→   523→      const [h, m] = raw.pickup_slot_start.split(':').map(Number)
   524→   524→      if (isNaN(h) || isNaN(m)) return false
   525→   525→      const slotMinutes = h * 60 + m
   526→   526→      return slotMinutes < windowStartMinutes
   527→   527→    }
   528→   528→
   529→   529→    // Map to typed response with proper snake_case -> camelCase conversion
   530→   530→    const mapped: AdminOrderWithItems[] = filteredOrders.map((o) => {
   531→   531→      const profile = profileMap.get(o.user_id)
   532→   532→      const branch = branchMap.get(o.branch_id)
   533→   533→      return {
   534→   534→        ...mapOrderRow(o),
   535→   535→        items: (o.order_items || []).map(mapOrderItemRow),
   536→   536→        branch: branch || null,
   537→   537→        customer: profile || { name: 'Customer', whatsappNumber: null, mobileNumber: null },
   538→   538→        isOverdue: status === 'confirmed' ? computeIsOverdue(o) : undefined,
   539→   539→      }
   540→   540→    })
   541→   541→
   542→   542→    return {
   543→   543→      success: true,
   544→   544→      orders: mapped,
   545→   545→      totalCount,
   546→   546→    }
   547→   547→  } catch (err) {
   548→   548→    console.error('[ADMIN ORDER SERVICE] listOrdersByStatus error:', err)
   549→   549→    return { success: false, orders: [], totalCount: 0, error: 'Failed to fetch orders' }
   550→   550→  }
   551→   551→}

   557→   557→export interface AcceptOrderResult {

   561→   561→  error?: string

   564→   564→/**
   565→   565→ * Accept an order: transition from 'confirmed' to 'accepted'.
   566→   566→ * - Validates order exists, is 'confirmed', and payment succeeded
   567→   567→ * - Prevents duplicate acceptance (idempotent by design)
   568→   568→ * - Records status change in order_status_history
   569→   569→ */
   570→   570→export async function acceptOrder(
   571→   571→  orderId: string,
   572→   572→  adminUserId: string
   573→   573→): Promise<AcceptOrderResult> {
   574→   574→  try {
   575→   575→    const supabase = createAdminClient()
   576→   576→
   577→   577→    // 1. Validate input
   578→   578→    if (!orderId || typeof orderId !== 'string') {
   579→   579→      return { success: false, message: 'Invalid order ID' }
   580→   580→    }
   581→   581→
   582→   582→    // 2. Fetch the order
   583→   583→    // eslint-disable-next-line @typescript-eslint/no-explicit-any
   584→   584→    const { data: order, error: fetchErr }: { data: RawOrderRow | null; error: any } = await supabase
   585→   585→      .from('orders')
   586→   586→      .select('*')
   587→   587→      .eq('id', orderId)
   588→   588→      .single()
   589→   589→
   590→   590→    if (fetchErr || !order) {
   591→   591→      return { success: false, message: 'Order not found' }
   592→   592→    }
   593→   593→
   594→   594→    // 3. Validate payment status — must be succeeded
   595→   595→    if (order.payment_status !== 'succeeded') {
   596→   596→      return { success: false, message: 'Cannot accept an unpaid order' }
   597→   597→    }
   598→   598→
   599→   599→    // 4. Validate current status
   600→   600→    if (order.order_status !== 'confirmed') {
   601→   601→      if (order.order_status === 'accepted') {
   602→   602→        return { success: true, message: 'Order already accepted' }
   603→   603→      }
   604→   604→      return { success: false, message: `Cannot accept order in '${order.order_status}' status` }
   605→   605→    }
   606→   606→
   607→   607→    // 5. Validate the transition is allowed
   608→   608→    const allowedTransitions = VALID_TRANSITIONS[order.order_status] || []
   609→   609→    if (!allowedTransitions.includes('accepted')) {
   610→   610→      return { success: false, message: 'Invalid status transition' }
   611→   611→    }
   612→   612→
   613→   613→    // 6. Update order status (race condition guard: only update if still confirmed)
   614→   614→    // eslint-disable-next-line @typescript-eslint/no-explicit-any
   615→   615→    const sb = supabase as any
   616→   616→    const { error: updateErr } = await sb
   617→   617→      .from('orders')
   618→   618→      .update({
   619→   619→        order_status: 'accepted',
   620→   620→        updated_at: new Date().toISOString(),
   621→   621→      })
   622→   622→      .eq('id', orderId)
   623→   623→      .eq('order_status', 'confirmed')
   624→   624→
   625→   625→    if (updateErr) {
   626→   626→      console.error('[ADMIN ORDER SERVICE] acceptOrder update error:', updateErr)
   627→   627→      return { success: false, message: 'Failed to update order. Please try again.' }
   628→   628→    }
   629→   629→
   630→   630→    // 7. Log status change in history
   631→   631→    //    Table schema: (order_id, status, note, created_at)
   632→   632→    //    Unique constraint: (order_id, status) — one row per status per order
   633→   633→    try {
   634→   634→      const { error: histErr } = await sb.from('order_status_history').insert({
   635→   635→        order_id: orderId,
   636→   636→        status: 'accepted',
   637→   637→        note: 'Accepted by admin',
   638→   638→      })
   639→   639→      // Ignore duplicate key errors (order already has 'accepted' status row)
   640→   640→      if (histErr && !String(histErr.message || histErr).includes('duplicate')) {
   641→   641→        console.error('[ADMIN ORDER SERVICE] Failed to log status history:', histErr)
   642→   642→      }
   643→   643→    } catch (historyErr) {
   644→   644→      // Log but don't fail — the order status was already updated
   645→   645→      console.error('[ADMIN ORDER SERVICE] Failed to log status history:', historyErr)
   646→   646→    }
   647→   647→
   648→   648→    return { success: true, message: 'Order accepted successfully' }
   649→   649→  } catch (err) {
   650→   650→    console.error('[ADMIN ORDER SERVICE] acceptOrder error:', err)
   651→   651→    return { success: false, message: 'Something went wrong. Please try again.' }
   652→   652→  }
   653→   653→}

   659→   659→/**
   660→   660→ * List all ongoing orders (accepted, preparing, ready_for_pickup).

   662→   662→ * Ordered by pickup time (soonest first), then by status priority.

   673→   673→    const offset = options?.offset || 0
   674→   674→
   675→   675→    // eslint-disable-next-line @typescript-eslint/no-explicit-any
   676→   676→    const sb = supabase as any
   677→   677→
   678→   678→    const selectFields = `
   679→   679→      *,
   680→   680→      order_items (*)
   681→   681→    `
   682→   682→
   683→   683→    let query = sb
   684→   684→      .from('orders')
   685→   685→      .select(selectFields, { count: 'exact' })
   686→   686→      .in('order_status', ['accepted', 'preparing', 'ready_for_pickup'])
   687→   687→      .order('pickup_slot_start', { ascending: true })
   688→   688→      .range(offset, offset + limit - 1)
   689→   689→
   690→   690→    if (options?.branchSlug) {
   691→   691→      query = query.eq('branch_id', options.branchSlug)

   693→   693→

   699→   699→    }
   700→   700→
   701→   701→    if (!orders || orders.length === 0) {
   702→   702→      return { success: true, orders: [], totalCount: 0 }
   703→   703→    }
   704→   704→
   705→   705→    const rawOrders = orders as RawOrderRow[]
   706→   706→

   708→   708→    // CRITICAL: profiles.id ≠ auth.users.id. Must join on auth_user_id.

   710→   710→    const { data: profiles } = await supabase
   711→   711→      .from('profiles')

   713→   713→      .in('auth_user_id', userIds)
   714→   714→

   716→   716→    for (const p of (profiles || []) as RawProfileRow[]) {
   717→   717→      profileMap.set(p.auth_user_id, {
   718→   718→        name: p.full_name || 'Customer',
   719→   719→        whatsappNumber: p.whatsapp_number || null,
   720→   720→        mobileNumber: p.mobile_number || null,
   721→   721→      })
   722→   722→    }
   723→   723→

   726→   726→    const { data: branchRows } = await supabase

   732→   732→    for (const b of (branchRows || []) as { id: string; name: string; slug: string; city: string }[]) {
   733→   733→      branchMap.set(b.id, { name: b.name, slug: b.slug, city: b.city })
   734→   734→    }
   735→   735→
   736→   736→    // --- Batch fetch accepted-at timestamps from status history ---
   737→   737→    const orderIds = rawOrders.map((o) => o.id)
   738→   738→    const { data: historyRows } = await supabase
   739→   739→      .from('order_status_history')
   740→   740→      .select('order_id, created_at')
   741→   741→      .in('order_id', orderIds)
   742→   742→      .eq('status', 'accepted')
   743→   743→
   744→   744→    const acceptedAtMap = new Map<string, string>()
   745→   745→    for (const row of (historyRows || []) as { order_id: string; created_at: string }[]) {
   746→   746→      if (!acceptedAtMap.has(row.order_id)) {
   747→   747→        acceptedAtMap.set(row.order_id, row.created_at)

   755→   755→      filteredOrders = filteredOrders.filter((o) => {
   756→   756→        const orderNum = (o.order_number || '').toLowerCase()
   757→   757→        const profile = profileMap.get(o.user_id)
   758→   758→        const customerName = (profile?.name || '').toLowerCase()
   759→   759→        return orderNum.includes(q) || customerName.includes(q)
   760→   760→      })
   761→   761→    }
   762→   762→
   763→   763→    // --- Status priority for sorting (accepted=1, preparing=2, ready=3) ---
   764→   764→    const statusWeight: Record<string, number> = {
   765→   765→      accepted: 1,
   766→   766→      preparing: 2,
   767→   767→      ready_for_pickup: 3,
   768→   768→    }

   804→   804→// ============================================================================
   805→   805→// UPDATE ORDER STATUS (Kitchen Workflow)
   806→   806→// ============================================================================

   808→   808→export interface UpdateOrderStatusResult {
   809→   809→  success: boolean
   810→   810→  message: string
   811→   811→  error?: string
   812→   812→}
   813→   813→
   814→   814→/**
   815→   815→ * Subset of VALID_TRANSITIONS for the kitchen workflow.
   816→   816→ * This module only handles forward transitions:
   817→   817→ *   accepted → preparing
   818→   818→ * preparing → ready_for_pickup is handled by the PIN generation endpoint.
   819→   819→ * Cancellation and other transitions are handled by other modules.
   820→   820→ */
   821→   821→const KITCHEN_TRANSITIONS: Record<string, AdminOrderStatus[]> = {
   822→   822→  accepted: ['preparing'],
   823→   823→}
   824→   824→
   825→   825→/**
   826→   826→ * Update an order's status as part of the kitchen workflow.
   827→   827→ * Handles: accepted → preparing.
   828→   828→ * preparing → ready_for_pickup is handled by the PIN generation endpoint.
   829→   829→ *
   830→   830→ * Security:
   831→   831→ * - Validates order exists and current DB status
   832→   832→ * - Validates transition is allowed
   833→   833→ * - Race condition guard: UPDATE only if order_status still equals current
   834→   834→ * - Records full audit trail in order_status_history
   835→   835→ *
   836→   836→ * @param orderId - UUID of the order
   837→   837→ * @param targetStatus - The new status to transition to
   838→   838→ * @param adminUserId - UUID of the admin making the change
   839→   839→ */
   840→   840→export async function updateOrderStatus(
   841→   841→  orderId: string,
   842→   842→  targetStatus: AdminOrderStatus,
   843→   843→  adminUserId: string
   844→   844→): Promise<UpdateOrderStatusResult> {
   845→   845→  try {
   846→   846→    const supabase = createAdminClient()
   847→   847→
   848→   848→    // 1. Validate inputs
   849→   849→    if (!orderId || typeof orderId !== 'string') {
   850→   850→      return { success: false, message: 'Invalid order ID' }
   851→   851→    }
   852→   852→    if (!targetStatus || typeof targetStatus !== 'string') {
   853→   853→      return { success: false, message: 'Invalid target status' }
   854→   854→    }
   855→   855→
   856→   856→    // 2. Validate target status is allowed in kitchen workflow
   857→   857→    const allowedTargets = Object.values(KITCHEN_TRANSITIONS).flat()
   858→   858→    if (!allowedTargets.includes(targetStatus)) {
   859→   859→      return { success: false, message: 'Invalid status transition for this workflow' }
   860→   860→    }
   861→   861→
   862→   862→    // 3. Fetch the current order
   863→   863→    // eslint-disable-next-line @typescript-eslint/no-explicit-any
   864→   864→    const { data: order, error: fetchErr }: { data: RawOrderRow | null; error: any } = await supabase
   865→   865→      .from('orders')
   866→   866→      .select('*')
   867→   867→      .eq('id', orderId)
   868→   868→      .single()
   869→   869→
   870→   870→    if (fetchErr || !order) {
   871→   871→      return { success: false, message: 'Order not found' }
   872→   872→    }
   873→   873→
   874→   874→    // 4. Determine what current status allows this transition
   875→   875→    const currentStatus = order.order_status
   876→   876→    const allowedFromCurrent = KITCHEN_TRANSITIONS[currentStatus]
   877→   877→    if (!allowedFromCurrent || !allowedFromCurrent.includes(targetStatus)) {
   878→   878→      // Provide a helpful message
   879→   879→      const statusLabels: Record<string, string> = {
   880→   880→        confirmed: 'Pending',
   881→   881→        accepted: 'Accepted',
   882→   882→        preparing: 'Preparing',
   883→   883→        ready_for_pickup: 'Ready for Pickup',
   884→   884→        completed: 'Completed',
   885→   885→        cancelled: 'Cancelled',
   886→   886→      }
   887→   887→      const from = statusLabels[currentStatus] || currentStatus
   888→   888→      const to = statusLabels[targetStatus] || targetStatus
   889→   889→      return { success: false, message: `Cannot move order from "${from}" to "${to}"` }
   890→   890→    }
   891→   891→
   892→   892→    // 5. Update with race condition guard (only update if status hasn't changed)
   893→   893→    // eslint-disable-next-line @typescript-eslint/no-explicit-any
   894→   894→    const sb = supabase as any
   895→   895→    const { error: updateErr, count: updateCount } = await sb
   896→   896→      .from('orders')
   897→   897→      .update({
   898→   898→        order_status: targetStatus,
   899→   899→        updated_at: new Date().toISOString(),
   900→   900→      })
   901→   901→      .eq('id', orderId)
   902→   902→      .eq('order_status', currentStatus)
   903→   903→
   904→   904→    if (updateErr) {
   905→   905→      console.error('[ADMIN ORDER SERVICE] updateOrderStatus update error:', updateErr)
   906→   906→      return { success: false, message: 'Failed to update order status. Please try again.' }
   907→   907→    }
   908→   908→
   909→   909→    // 6. Check if the update actually happened (0 rows = race condition / already changed)
   910→   910→    if (updateCount === 0) {
   911→   911→      // Re-fetch to get current status for a helpful message
   912→   912→      const { data: freshOrder }: { data: any } = await sb
   913→   913→        .from('orders')
   914→   914→        .select('order_status')
   915→   915→        .eq('id', orderId)
   916→   916→        .single()
   917→   917→
   918→   918→      const statusLabels: Record<string, string> = {
   919→   919→        accepted: 'Accepted', preparing: 'Preparing',
   920→   920→        ready_for_pickup: 'Ready for Pickup', completed: 'Completed',
   921→   921→        cancelled: 'Cancelled', confirmed: 'Pending',
   922→   922→      }
   923→   923→      const current = statusLabels[freshOrder?.order_status] || freshOrder?.order_status || 'Unknown'
   924→   924→      return { success: false, message: `Order status was already updated to "${current}" by another action` }
   925→   925→    }
   926→   926→
   927→   927→    // 7. Log status change in history
   928→   928→    //    Table schema: (order_id, status, note, created_at)
   929→   929→    //    Unique constraint: (order_id, status) — one row per status per order
   930→   930→    //    Note encodes: previous → new | admin_id
